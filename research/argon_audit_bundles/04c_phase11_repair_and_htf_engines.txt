"""Ophelia Phase 11: Deterministic Strategy Repair & Regime / Geometry Discovery Engine.

Layer: control_plane/phase11_repair_engine.py
Contract:
- Authoritative investigation of the failure modes of the raw ENTRY_READY retest strategy.
- Answers: "Under what observable market conditions does an existing ENTRY_READY retest setup
  have positive net expectancy, and under what conditions does it systematically fail?"
- Freezes existing ENTRY_READY strategy as BASELINE_A.
- Computes comprehensive feature snapshots (Structure, Volume, Volatility, Geometry, Regime) for every candidate.
- Implements deterministic Failure Taxonomy:
    FALSE_RETEST, RANGE_CHOP, INSUFFICIENT_FOLLOW_THROUGH, OVEREXTENDED_ENTRY,
    LOW_PARTICIPATION, VOLATILITY_COLLAPSE, TARGET_TOO_FAR, STOP_TOO_TIGHT,
    STRUCTURAL_INVALIDATION, SUCCESS.
- Constructs and evaluates controlled variants:
    Variant A: BASELINE_A (Unfiltered)
    Variant B: BASELINE + REGIME COMPATIBILITY (Exclude RANGE_CHOP)
    Variant C: BASELINE + GEOMETRY VIABILITY (Filter unviable R:R and tight SL)
    Variant D: BASELINE + STRUCTURE QUALITY (Displacement >= 1.5 ATR, Rejection body >= 0.5 range, RVOL >= 1.2)
    Variant E: BASELINE + REGIME + GEOMETRY
    Variant F: BASELINE + REGIME + GEOMETRY + STRUCTURE QUALITY
- Strict Chronological Partitions (Oct 1 to Nov 30, 2023, 61 days):
    TRAIN:      Oct 01 to Oct 20 (Exploratory fitting)
    VALIDATION: Oct 21 to Nov 05 (Variant selection)
    TEST:       Nov 06 to Nov 30 (Untouched Out-of-Sample)
- Friction Sensitivity: Zero friction, Low (0.05%), Base (0.09%), High (0.14%).
- Clustered Bootstrap 95% CIs and Multiple Testing Correction (Benjamini-Hochberg FDR).
- Invariants: TRADING_AUTHORITY = FALSE, EXECUTION_MODE = SHADOW, LIVE_ORDERS = FORBIDDEN.
- TimesFM ISOLATED: Not used in candidate evaluation for Phase 11.
"""

# Friction regimes (Round-trip fees + Round-trip slippage)

FRICTION_BASE = 0.0009  # 0.05% fee + 0.04% slippage = 9.0 bps (Phase 10.3 standard)
FRICTION_HIGH = 0.0014  # 0.08% fee + 0.06% slippage = 14.0 bps

def assign_split(ts: int) -> str:
    if TS_OCT_01_2023 <= ts <= (TS_OCT_20_2023 - EMBARGO_MS):
        return "TRAIN"
    elif (TS_OCT_20_2023 - EMBARGO_MS) < ts < TS_OCT_21_2023:
        return "EMBARGO"
    elif TS_OCT_21_2023 <= ts <= (TS_NOV_05_2023 - EMBARGO_MS):
        return "VALIDATION"
    elif (TS_NOV_05_2023 - EMBARGO_MS) < ts < TS_NOV_06_2023:
        return "EMBARGO"
    elif TS_NOV_06_2023 <= ts <= TS_NOV_30_2023:
        return "TEST"
    return "OUT_OF_BOUNDS"


def classify_rr_bucket(rr: float) -> str:
    if rr < 1.25:
        return "<1.25"
    elif rr < 1.5:
        return "1.25-1.5"
    elif rr < 2.0:
        return "1.5-2.0"
    elif rr < 2.5:
        return "2.0-2.5"
    elif rr < 3.0:
        return "2.5-3.0"
    else:
        return ">3.0"


def classify_deterministic_regime(
    ts: int,
    close: float,
    ema_20: float,
    ema_50: float,
    atr_pct: float,
    adx_proxy: float
) -> str:
    """Classifies market regime strictly timestamp-causally."""
    dt = datetime.datetime.fromtimestamp(ts / 1000, tz=datetime.timezone.utc)
    # Major ETF anticipation shock
    if datetime.date(2023, 10, 16) <= dt.date() <= datetime.date(2023, 10, 26) or atr_pct > 0.015:
        return "VOLATILITY_SHOCK"
    
    if adx_proxy < 18.0 or abs(close - ema_20) / max(1e-6, close) < 0.0015:
        return "RANGE_CHOP"
    elif (close > ema_20 > ema_50) or (close < ema_20 < ema_50):
        if atr_pct >= 0.006:
            return "TREND_EXPANSION"
        else:
            return "TREND_DECELERATION"
    else:
        return "RANGE_CHOP"


def classify_failure_taxonomy(
    exit_reason: str,
    direction: str,
    entry: float,
    sl: float,
    tp1: float,
    mfe: float,
    mae: float,
    cur_atr: float,
    regime: str,
    rvol: float,
    displacement_atr: float,
    sl_atr: float,
    tp1_atr: float,
    retest_depth_atr: float
) -> str:
    """Assigns deterministic taxonomy to candidate outcome."""
    if exit_reason in ("TP1", "TP2"):
        return "SUCCESS"
        
    # Check distinct failure drivers in priority order
    if regime == "RANGE_CHOP":
        return "RANGE_CHOP"
    if sl_atr < 0.4:
        return "STOP_TOO_TIGHT"
    if tp1_atr > 3.0:
        return "TARGET_TOO_FAR"
    if retest_depth_atr > 1.2:
        return "FALSE_RETEST"
    if rvol < 1.0:
        return "LOW_PARTICIPATION"
    if displacement_atr > 2.5 and (abs(entry - (entry - mfe)) / cur_atr) > 2.0:
        return "OVEREXTENDED_ENTRY"
    if abs(mfe) < (0.4 * cur_atr):
        return "INSUFFICIENT_FOLLOW_THROUGH"
    if abs(mae) >= (0.95 * abs(entry - sl)):
        return "STRUCTURAL_INVALIDATION"
        
    return "UNKNOWN"


def bootstrap_ci(
    values: List[float],
    n_iterations: int = 2000,
    alpha: float = 0.05
) -> Tuple[float, float, float]:
    """Computes mean and two-sided bootstrap confidence interval [lower, upper]."""
    if not values:
        return 0.0, 0.0, 0.0
    arr = np.array(values, dtype=np.float64)
    sample_mean = float(np.mean(arr))
    if len(arr) == 1:
        return sample_mean, sample_mean, sample_mean
        
    rng = np.random.default_rng(42)
    boot_means = np.zeros(n_iterations, dtype=np.float64)
    n = len(arr)
    for i in range(n_iterations):
        resample = rng.choice(arr, size=n, replace=True)
        boot_means[i] = np.mean(resample)
        
    lower = float(np.percentile(boot_means, (alpha / 2.0) * 100.0))
    upper = float(np.percentile(boot_means, (1.0 - alpha / 2.0) * 100.0))
    return sample_mean, lower, upper


class Phase11StrategyRepairRunner:
    """Executes the Phase 11 deterministic strategy repair and failure analysis."""

    def __init__(self, parquet_path: str = "data/processed/binance_um_universe_1m_v2.parquet"):
        self.parquet_path = parquet_path
        self.symbols = ["BTCUSDT", "ETHUSDT", "SOLUSDT"]
        self.snapshots: List[CandidateFeatureSnapshot] = []
        self.results: Dict[str, Any] = {}

    def run_analysis(self) -> Dict[str, Any]:
        logger.info("Starting Phase 11 Deterministic Strategy Repair & Regime Discovery...")
        t0 = time.perf_counter()
        
        # 1. Load data and generate candidates
        from .phase10_3_validation_engine import load_and_resample_5m
        candles_by_symbol = load_and_resample_5m(
            self.parquet_path,
            t_start=TS_OCT_01_2023,
            t_end=TS_NOV_30_2023,
            symbols=self.symbols
        )
        
        # 2. Extract feature snapshots for all candidates
        for sym in self.symbols:
            c5m = candles_by_symbol.get(sym, [])
            if not c5m:
                continue
            cands = generate_retest_candidates(c5m, sym, timeframe=5, context_lookback=128)
            for cand in cands:
                snap = self._extract_candidate_snapshot(cand, c5m)
                if snap:
                    self.snapshots.append(snap)
                    
        # Sort chronologically
        self.snapshots.sort(key=lambda s: s.timestamp_ms)
        logger.info("Total feature snapshots generated: %d", len(self.snapshots))
        
        # 3. Compute baseline performance, regime matrix, geometry matrix, failure taxonomy, variants
        self.results = self._compute_research_matrices()
        
        elapsed = time.perf_counter() - t0
        self.results["elapsed_seconds"] = round(elapsed, 2)
        logger.info("Phase 11 analysis complete in %.2f seconds", elapsed)
        return self.results

    def _extract_candidate_snapshot(
        self,
        cand: Dict[str, Any],
        symbol_candles: List[dict]
    ) -> Optional[CandidateFeatureSnapshot]:
        ts = int(cand["timestamp_ms"])
        cand_id = cand["candidate_id"]
        sym = cand["symbol"]
        direction = cand["direction"].upper()
        entry = float(cand["entry_price"])
        sl = float(cand["sl_price"])
        tp1 = float(cand["tp1_price"])
        tp2 = float(cand["tp2_price"]) if cand.get("tp2_price") else None
        
        split = assign_split(ts)
        dt_str = datetime.datetime.fromtimestamp(ts / 1000, tz=datetime.timezone.utc).isoformat()
        
        risk = (entry - sl) if direction == "LONG" else (sl - entry)
        if risk <= 0:
            return None
        reward1 = (tp1 - entry) if direction == "LONG" else (entry - tp1)
        raw_rr = reward1 / risk
        rr_bucket = classify_rr_bucket(raw_rr)
        
        # Historical context (bars <= ts)
        hist_candles = [c for c in symbol_candles if int(c["ts"]) <= ts]
        if len(hist_candles) < 30:
            return None
            
        future_candles = [c for c in symbol_candles if int(c["ts"]) > ts][:12]
        if len(future_candles) < 3:
            return None
            
        # Structure & Volatility indicators from history
        highs = [float(c["high"]) for c in hist_candles]
        lows = [float(c["low"]) for c in hist_candles]
        closes = [float(c["close"]) for c in hist_candles]
        vols = [float(c.get("volume") or 0.0) for c in hist_candles]
        
        atr_series = calculate_atr(highs, lows, closes, 14)
        cur_atr = max(1e-6, atr_series[-1] if atr_series else entry * 0.01)
        atr_pct = cur_atr / entry
        
        # Volatility regime
        if atr_pct < 0.004:
            vol_regime = "LOW"
        elif atr_pct > 0.010:
            vol_regime = "HIGH"
        else:
            vol_regime = "NORMAL"
            
        # EMAs
        ema_20 = float(pd.Series(closes).ewm(span=20, adjust=False).mean().iloc[-1])
        ema_50 = float(pd.Series(closes).ewm(span=50, adjust=False).mean().iloc[-1])
        trend_state = "BULLISH" if ema_20 > ema_50 else ("BEARISH" if ema_20 < ema_50 else "NEUTRAL")
        
        # ADX Proxy: mean absolute directional move over 14 bars normalized by ATR
        diffs = [abs(closes[i] - closes[i-1]) for i in range(len(closes)-14, len(closes))]
        adx_proxy = min(50.0, (sum(diffs) / (14.0 * cur_atr)) * 25.0)
        
        # Realized volatility (annualized from 20-bar log returns)
        log_rets = [math.log(closes[i] / closes[i-1]) for i in range(len(closes)-20, len(closes))]
        realized_vol = float(np.std(log_rets) * math.sqrt(288.0 * 365.0))
        
        # Regime classification
        regime = classify_deterministic_regime(ts, entry, ema_20, ema_50, atr_pct, adx_proxy)
        
        # Retest & Rejection details
        rejection_bar = hist_candles[-1]
        r_open = float(rejection_bar["open"])
        r_high = float(rejection_bar["high"])
        r_low = float(rejection_bar["low"])
        r_close = float(rejection_bar["close"])
        r_vol = float(rejection_bar.get("volume") or 0.0)
        
        rej_range = max(1e-6, r_high - r_low)
        rej_body = abs(r_close - r_open)
        rej_body_ratio = rej_body / rej_range
        rej_close_loc = (r_close - r_low) / rej_range if direction == "LONG" else (r_high - r_close) / rej_range
        
        # Volume metrics
        prev_vols = vols[-21:-1]
        avg_vol = float(np.mean(prev_vols)) if prev_vols else r_vol
        rvol = r_vol / max(1e-6, avg_vol)
        vol_expansion = r_vol / max(1e-6, vols[-2] if len(vols) >= 2 else avg_vol)
        taker_imbalance = 0.0  # Placeholder if taker not separated in 5m
        
        # Structure measurements
        breakout_px = float(cand.get("breakout_price") or entry)
        retest_touch_px = float(cand.get("retest_touch_price") or entry)
        retest_elapsed = int(cand.get("retest_elapsed_bars") or 1)
        
        disp_mag = abs(entry - breakout_px)
        disp_atr = disp_mag / cur_atr
        retest_depth = abs(entry - retest_touch_px)
        retest_depth_atr = retest_depth / cur_atr
        
        tp1_atr = reward1 / cur_atr
        sl_atr = risk / cur_atr
        friction_sl_ratio = (FRICTION_BASE * entry) / risk
        friction_tp_ratio = (FRICTION_BASE * entry) / reward1
        
        # Forward Trajectory Evaluation (Conservative same-candle SL ordering)
        f_highs = [float(b["high"]) for b in future_candles]
        f_lows = [float(b["low"]) for b in future_candles]
        f_closes = [float(b["close"]) for b in future_candles]
        
        exit_reason = "HORIZON_EXPIRED"
        time_to_exit_min = len(future_candles) * 5
        exit_price = f_closes[-1]
        
        if direction == "LONG":
            mfe = max(f_highs) - entry
            mae = min(f_lows) - entry
            for idx, b in enumerate(future_candles):
                bh, bl = float(b["high"]), float(b["low"])
                if bl <= sl:
                    exit_reason = "SL"
                    time_to_exit_min = (idx + 1) * 5
                    exit_price = sl
                    break
                elif bh >= tp1:
                    exit_reason = "TP1"
                    time_to_exit_min = (idx + 1) * 5
                    exit_price = tp1
                    break
            gross_r = (exit_price - entry) / risk
        else:
            mfe = entry - min(f_lows)
            mae = entry - max(f_highs)
            for idx, b in enumerate(future_candles):
                bh, bl = float(b["high"]), float(b["low"])
                if bh >= sl:
                    exit_reason = "SL"
                    time_to_exit_min = (idx + 1) * 5
                    exit_price = sl
                    break
                elif bl <= tp1:
                    exit_reason = "TP1"
                    time_to_exit_min = (idx + 1) * 5
                    exit_price = tp1
                    break
            gross_r = (entry - exit_price) / risk
            
        cost_r_base = (FRICTION_BASE * entry) / risk
        cost_r_low = (FRICTION_LOW * entry) / risk
        cost_r_high = (FRICTION_HIGH * entry) / risk
        
        net_r_base = gross_r - cost_r_base
        net_r_low = gross_r - cost_r_low
        net_r_high = gross_r - cost_r_high
        
        # Taxonomy classification
        failure_cat = classify_failure_taxonomy(
            exit_reason=exit_reason,
            direction=direction,
            entry=entry,
            sl=sl,
            tp1=tp1,
            mfe=mfe,
            mae=mae,
            cur_atr=cur_atr,
            regime=regime,
            rvol=rvol,
            displacement_atr=disp_atr,
            sl_atr=sl_atr,
            tp1_atr=tp1_atr,
            retest_depth_atr=retest_depth_atr
        )
        
        return CandidateFeatureSnapshot(
            candidate_id=cand_id,
            symbol=sym,
            direction=direction,
            timestamp_ms=ts,
            dt_utc=dt_str,
            split=split,
            entry_price=entry,
            sl_price=sl,
            tp1_price=tp1,
            tp2_price=tp2,
            risk_amount=risk,
            reward_tp1=reward1,
            raw_rr=raw_rr,
            rr_bucket=rr_bucket,
            tp1_atr=tp1_atr,
            sl_atr=sl_atr,
            friction_sl_ratio=friction_sl_ratio,
            friction_tp_ratio=friction_tp_ratio,
            breakout_price=breakout_px,
            retest_touch_price=retest_touch_px,
            retest_elapsed_bars=retest_elapsed,
            displacement_mag=disp_mag,
            displacement_atr=disp_atr,
            retest_depth=retest_depth,
            retest_depth_atr=retest_depth_atr,
            rejection_range=rej_range,
            rejection_body_ratio=rej_body_ratio,
            rejection_close_loc=rej_close_loc,
            rejection_volume=r_vol,
            rvol=rvol,
            volume_expansion=vol_expansion,
            taker_imbalance=taker_imbalance,
            cur_atr=cur_atr,
            atr_pct=atr_pct,
            realized_vol=realized_vol,
            volatility_regime=vol_regime,
            ema_20=ema_20,
            ema_50=ema_50,
            trend_state=trend_state,
            adx_proxy=adx_proxy,
            regime=regime,
            exit_reason=exit_reason,
            time_to_exit_min=time_to_exit_min,
            gross_realized_r=gross_r,
            net_realized_r_base=net_r_base,
            net_realized_r_low=net_r_low,
            net_realized_r_high=net_r_high,
            mfe=mfe,
            mae=mae,
            mfe_atr=mfe / cur_atr,
            mae_atr=mae / cur_atr,
            failure_category=failure_cat
        )

    def _compute_research_matrices(self) -> Dict[str, Any]:
        snaps = self.snapshots
        
        def cohort_stats(subset: List[CandidateFeatureSnapshot], friction_mode="base") -> Dict[str, Any]:
            if not subset:
                return {
                    "count": 0, "win_rate": 0.0, "net_r": 0.0, "gross_r": 0.0,
                    "mean_r": 0.0, "median_r": 0.0, "pf": 0.0, "ci_mean_r": [0.0, 0.0], "ci_wr": [0.0, 0.0]
                }
            n = len(subset)
            wins = sum(1 for s in subset if s.exit_reason in ("TP1", "TP2"))
            wr = (wins / n) * 100.0
            
            if friction_mode == "low":
                net_rs = [s.net_realized_r_low for s in subset]
            elif friction_mode == "high":
                net_rs = [s.net_realized_r_high for s in subset]
            elif friction_mode == "zero":
                net_rs = [s.gross_realized_r for s in subset]
            else:
                net_rs = [s.net_realized_r_base for s in subset]
                
            tot_net = sum(net_rs)
            tot_gross = sum(s.gross_realized_r for s in subset)
            mean_r = float(np.mean(net_rs))
            med_r = float(np.median(net_rs))
            
            g_wins = sum(s.gross_realized_r for s in subset if s.gross_realized_r > 0)
            g_losses = abs(sum(s.gross_realized_r for s in subset if s.gross_realized_r < 0))
            pf = round(g_wins / max(1e-6, g_losses), 2)
            
            _, ci_l, ci_u = bootstrap_ci(net_rs)
            wr_bool = [1.0 if s.exit_reason in ("TP1", "TP2") else 0.0 for s in subset]
            _, wr_l, wr_u = bootstrap_ci(wr_bool)
            
            return {
                "count": n,
                "win_rate": round(wr, 2),
                "wins": wins,
                "losses": n - wins,
                "gross_r": round(tot_gross, 2),
                "net_r": round(tot_net, 2),
                "mean_r": round(mean_r, 3),
                "median_r": round(med_r, 3),
                "pf": pf,
                "ci_mean_r": [round(ci_l, 3), round(ci_u, 3)],
                "ci_wr": [round(wr_l * 100.0, 2), round(wr_u * 100.0, 2)]
            }
            
        train_snaps = [s for s in snaps if s.split == "TRAIN"]
        val_snaps = [s for s in snaps if s.split == "VALIDATION"]
        test_snaps = [s for s in snaps if s.split == "TEST"]
        
        # 1. Baseline Performance
        baseline_stats = {
            "all": cohort_stats(snaps),
            "train": cohort_stats(train_snaps),
            "val": cohort_stats(val_snaps),
            "test": cohort_stats(test_snaps)
        }

                }

        variants_eval = {}

        var_tests.sort(key=lambda x: x["p_value"])

                item["status"] = "OUT-OF-SAMPLE SUPPORTED"
            elif item["delta_mean_r"] > 0:
                item["status"] = "PROMISING (RESEARCH HYPOTHESIS)"
            else:
                item["status"] = "NOT SUPPORTED"

        return {
            "dataset_summary": {
                "period": "2023-10-01 to 2023-11-30 (61 days / 8.7 weeks)",
                "total_candidates": len(snaps),
                "train_count": len(train_snaps),

            },
            "baseline_performance": baseline_stats,
            "failure_taxonomy": failure_matrix,
            "regime_analysis": regime_matrix,
            "geometry_analysis": rr_matrix,
            "structure_quality": structure_quality,
            "candidate_clustering": clustering_matrix,
            "friction_sensitivity": friction_matrix,
            "asset_comparison": asset_matrix,
            "strategy_variants": variants_eval,
            "multiple_testing_variants": var_tests
        }

def resample_htf_candles(candles_5m: List[dict], bar_ms: int) -> List[dict]:
    """Resamples sorted 5m candles into closed higher timeframe candles.
    
    bar_ms: 3600 * 1000 for 1H, 4 * 3600 * 1000 for 4H.
    A bar covering [T, T + bar_ms) is stamped at T + bar_ms (its close time).
    """
    if not candles_5m:
        return []
        
    grouped: Dict[int, List[dict]] = {}
    for c in candles_5m:
        ts = int(c["ts"])
        # Bucket key is start time of HTF bar
        bucket_start = (ts // bar_ms) * bar_ms
        if bucket_start not in grouped:
            grouped[bucket_start] = []
        grouped[bucket_start].append(c)
        
    htf_candles = []
    # Expected number of 5m bars per HTF bar
    expected_bars = bar_ms // (5 * 60 * 1000)
    
    for bucket_start, bars in sorted(grouped.items(), key=lambda x: x[0]):
        # Only accept completed or nearly completed bars (>= 90% bars present)
        if len(bars) < max(1, int(expected_bars * 0.9)):
            continue
            
        b_open = float(bars[0]["open"])
        b_high = max(float(b["high"]) for b in bars)
        b_low = min(float(b["low"]) for b in bars)
        b_close = float(bars[-1]["close"])
        b_vol = sum(float(b.get("volume") or 0.0) for b in bars)
        close_ts = bucket_start + bar_ms
        
        htf_candles.append({
            "ts": close_ts,
            "open": b_open,
            "high": b_high,
            "low": b_low,
            "close": b_close,
            "volume": b_vol,
        })
        
    return htf_candles


def compute_htf_features(
    cand_ts: int,
    direction: str,
    candles_1h: List[dict],
    candles_4h: List[dict]
) -> Dict[str, Any]:
    """Extracts strictly causal HTF features using only bars closed BEFORE cand_ts."""
    # Strict causal slicing: bars with close_ts <= cand_ts
    hist_1h = [c for c in candles_1h if int(c["ts"]) <= cand_ts]
    hist_4h = [c for c in candles_4h if int(c["ts"]) <= cand_ts]
    
    # 1. 1H Features
    if len(hist_1h) >= 20:
        c1h = [float(c["close"]) for c in hist_1h]
        h1h = [float(c["high"]) for c in hist_1h]
        l1h = [float(c["low"]) for c in hist_1h]
        
        ema_20_1h = float(pd.Series(c1h).ewm(span=20, adjust=False).mean().iloc[-1])
        ema_50_1h = float(pd.Series(c1h).ewm(span=50, adjust=False).mean().iloc[-1])
        
        if ema_20_1h > ema_50_1h * 1.0005:
            trend_1h = "BULLISH"
        elif ema_20_1h < ema_50_1h * 0.9995:
            trend_1h = "BEARISH"
        else:
            trend_1h = "NEUTRAL"
            
        trend_aligned_1h = (direction == "LONG" and trend_1h == "BULLISH") or (direction == "SHORT" and trend_1h == "BEARISH")
        
        atr_1h_series = calculate_atr(h1h, l1h, c1h, 14)
        cur_atr_1h = atr_1h_series[-1] if atr_1h_series else c1h[-1] * 0.01
        atr_pct_1h = cur_atr_1h / max(1e-6, c1h[-1])
        
        if atr_pct_1h < 0.006:
            vol_regime_1h = "LOW"
        elif atr_pct_1h > 0.018:
            vol_regime_1h = "HIGH"
        else:
            vol_regime_1h = "NORMAL"
            
        diffs_1h = [abs(c1h[i] - c1h[i-1]) for i in range(max(1, len(c1h)-10), len(c1h))]
        adx_proxy_1h = (sum(diffs_1h) / max(1e-6, len(diffs_1h) * cur_atr_1h)) * 25.0
        expansion_1h = "EXPANSION" if adx_proxy_1h >= 18.0 else "RANGE"
        
        # Structure location: 48-hour swing range
        window_1h = hist_1h[-48:]
        high_48h = max(float(b["high"]) for b in window_1h)
        low_48h = min(float(b["low"]) for b in window_1h)
        rng_48h = max(1e-6, high_48h - low_48h)
        swing_loc_1h = (c1h[-1] - low_48h) / rng_48h
        
        # Favorable structure: not buying into extreme ceiling (>0.90) or selling into floor (<0.10)
        structure_favorable_1h = (direction == "LONG" and swing_loc_1h <= 0.85) or (direction == "SHORT" and swing_loc_1h >= 0.15)
        has_1h = True
    else:
        trend_1h = "NEUTRAL"
        trend_aligned_1h = False
        atr_pct_1h = 0.01
        vol_regime_1h = "NORMAL"
        adx_proxy_1h = 15.0
        expansion_1h = "RANGE"
        swing_loc_1h = 0.5
        structure_favorable_1h = True
        has_1h = False

    # 2. 4H Features
    if len(hist_4h) >= 10:
        c4h = [float(c["close"]) for c in hist_4h]
        h4h = [float(c["high"]) for c in hist_4h]
        l4h = [float(c["low"]) for c in hist_4h]
        
        ema_20_4h = float(pd.Series(c4h).ewm(span=20, adjust=False).mean().iloc[-1])
        ema_50_4h = float(pd.Series(c4h).ewm(span=50, adjust=False).mean().iloc[-1])
        
        if ema_20_4h > ema_50_4h * 1.001:
            trend_4h = "BULLISH"
        elif ema_20_4h < ema_50_4h * 0.999:
            trend_4h = "BEARISH"
        else:
            trend_4h = "NEUTRAL"
            
        trend_aligned_4h = (direction == "LONG" and trend_4h == "BULLISH") or (direction == "SHORT" and trend_4h == "BEARISH")
        
        atr_4h_series = calculate_atr(h4h, l4h, c4h, min(14, len(c4h)-1))
        cur_atr_4h = atr_4h_series[-1] if atr_4h_series else c4h[-1] * 0.02
        atr_pct_4h = cur_atr_4h / max(1e-6, c4h[-1])
        vol_regime_4h = "HIGH" if atr_pct_4h > 0.030 else ("LOW" if atr_pct_4h < 0.012 else "NORMAL")
        has_4h = True
    else:
        trend_4h = "NEUTRAL"
        trend_aligned_4h = False
        atr_pct_4h = 0.02
        vol_regime_4h = "NORMAL"
        has_4h = False

    # Combined HTF alignment: 1H trend alignment + favorable structure location
    htf_aligned = bool(trend_aligned_1h and structure_favorable_1h)
    
    return {
        "has_1h": has_1h,
        "trend_1h": trend_1h,
        "trend_aligned_1h": trend_aligned_1h,
        "atr_pct_1h": atr_pct_1h,
        "vol_regime_1h": vol_regime_1h,
        "adx_proxy_1h": adx_proxy_1h,
        "expansion_state_1h": expansion_1h,
        "swing_loc_1h": swing_loc_1h,
        "structure_favorable_1h": structure_favorable_1h,
        "has_4h": has_4h,
        "trend_4h": trend_4h,
        "trend_aligned_4h": trend_aligned_4h,
        "atr_pct_4h": atr_pct_4h,
        "vol_regime_4h": vol_regime_4h,
        "htf_aligned": htf_aligned,
    }


def simulate_trade_management(
    policy_name: str,
    direction: str,
    entry: float,
    sl: float,
    tp1: float,
    tp2: Optional[float],
    future_candles_5m: List[dict],
    cur_atr: float,
    friction_rate: float = FRICTION_BASE
) -> Tuple[float, float, str, int]:
    """Simulates a specific trade management policy forward through 5m candles.
    
    Returns:
        (gross_realized_r, net_realized_r, exit_reason, exit_duration_min)
    """
    risk = abs(entry - sl)
    if risk <= 0:
        return 0.0, 0.0, "INVALID_RISK", 0
        
    reward1 = abs(tp1 - entry)
    r_target_tp1 = reward1 / risk
    tp2_price = tp2 if tp2 is not None else (entry + 2.0 * reward1 if direction == "LONG" else entry - 2.0 * reward1)
    
    cost_in_r = (friction_rate * entry) / risk
    
    if not future_candles_5m:
        return 0.0, -cost_in_r, "NO_DATA", 0
        
    # State variables
    cur_sl = sl
    be_active = False
    partial_taken = False
    realized_r_accum = 0.0
    active_size = 1.0  # Fraction of initial position currently open
    
    highest_high = entry
    lowest_low = entry
    
    exit_reason = "HORIZON_EXPIRED"
    exit_bars = len(future_candles_5m)
    final_exit_price = float(future_candles_5m[-1]["close"])
    
    max_bars = 24  # Max 2-hour monitoring window
    monitored_bars = future_candles_5m[:max_bars]
    
    for bar_idx, bar in enumerate(monitored_bars, 1):
        bh = float(bar["high"])
        bl = float(bar["low"])
        bc = float(bar["close"])
        
        # Update extreme points for trailing
        if bh > highest_high:
            highest_high = bh
        if bl < lowest_low:
            lowest_low = bl
            
        current_mfe_r = (highest_high - entry) / risk if direction == "LONG" else (entry - lowest_low) / risk
        
        # Policy-specific pre-check stop adjustments
        if policy_name == "BE_0.5R" and not be_active and current_mfe_r >= 0.5:
            cur_sl = entry
            be_active = True
        elif policy_name == "BE_0.8R" and not be_active and current_mfe_r >= 0.8:
            cur_sl = entry
            be_active = True
        elif policy_name == "BE_1.0R" and not be_active and current_mfe_r >= 1.0:
            cur_sl = entry
            be_active = True
        elif policy_name == "ATR_TRAILING":
            trail_dist = 1.5 * cur_atr
            if direction == "LONG":
                new_stop = highest_high - trail_dist
                if new_stop > cur_sl:
                    cur_sl = new_stop
            else:
                new_stop = lowest_low + trail_dist
                if new_stop < cur_sl:
                    cur_sl = new_stop
        elif policy_name == "STRUCTURAL_TRAILING" and bar_idx >= 3:
            # Trailing stop anchored to prior 2-bar low/high
            if direction == "LONG":
                prev_low = min(float(monitored_bars[b]["low"]) for b in range(max(0, bar_idx-3), bar_idx))
                if prev_low > cur_sl:
                    cur_sl = prev_low
            else:
                prev_high = max(float(monitored_bars[b]["high"]) for b in range(max(0, bar_idx-3), bar_idx))
                if prev_high < cur_sl:
                    cur_sl = prev_high
                    
        # Policy 9: Time-based exit at 6 bars (30m)
        if policy_name == "TIME_EXIT_30M" and bar_idx == 6:
            exit_reason = "TIME_EXIT_30M"
            exit_bars = bar_idx
            final_exit_price = bc
            break

        # Check Stop and Target triggers on this candle
        if direction == "LONG":
            hit_stop = (bl <= cur_sl)
            hit_tp1 = (bh >= tp1)
            hit_tp2 = (bh >= tp2_price)
            
            # Conservative rule: if stop touched, exit triggered
            if hit_stop:
                exit_price = cur_sl
                r_step = (exit_price - entry) / risk
                realized_r_accum += active_size * r_step
                active_size = 0.0
                exit_reason = "BE_EXIT" if be_active and abs(cur_sl - entry) < 1e-4 else "SL"
                exit_bars = bar_idx
                break
                
            if hit_tp1 and not partial_taken:
                if policy_name in ("TP1_PARTIAL", "PARTIAL_TRAILING"):
                    # Close 50% at TP1
                    r_tp1 = (tp1 - entry) / risk
                    realized_r_accum += 0.5 * r_tp1
                    active_size = 0.5
                    partial_taken = True
                    # Move stop to breakeven
                    cur_sl = entry
                    be_active = True
                    if policy_name == "PARTIAL_TRAILING":
                        # Trail remaining 50%
                        pass
                elif policy_name == "FIXED_SL_TP":
                    r_tp1 = (tp1 - entry) / risk
                    realized_r_accum += active_size * r_tp1
                    active_size = 0.0
                    exit_reason = "TP1"
                    exit_bars = bar_idx
                    break
                    
            if partial_taken and hit_tp2:
                r_tp2 = (tp2_price - entry) / risk
                realized_r_accum += active_size * r_tp2
                active_size = 0.0
                exit_reason = "TP2"
                exit_bars = bar_idx
                break

        else: # SHORT
            hit_stop = (bh >= cur_sl)
            hit_tp1 = (bl <= tp1)
            hit_tp2 = (bl <= tp2_price)
            
            if hit_stop:
                exit_price = cur_sl
                r_step = (entry - exit_price) / risk
                realized_r_accum += active_size * r_step
                active_size = 0.0
                exit_reason = "BE_EXIT" if be_active and abs(cur_sl - entry) < 1e-4 else "SL"
                exit_bars = bar_idx
                break
                
            if hit_tp1 and not partial_taken:
                if policy_name in ("TP1_PARTIAL", "PARTIAL_TRAILING"):
                    r_tp1 = (entry - tp1) / risk
                    realized_r_accum += 0.5 * r_tp1
                    active_size = 0.5
                    partial_taken = True
                    cur_sl = entry
                    be_active = True
                elif policy_name == "FIXED_SL_TP":
                    r_tp1 = (entry - tp1) / risk
                    realized_r_accum += active_size * r_tp1
                    active_size = 0.0
                    exit_reason = "TP1"
                    exit_bars = bar_idx
                    break
                    
            if partial_taken and hit_tp2:
                r_tp2 = (entry - tp2_price) / risk
                realized_r_accum += active_size * r_tp2
                active_size = 0.0
                exit_reason = "TP2"
                exit_bars = bar_idx
                break

    # If position still partially or fully open at horizon end, close at final candle close
    if active_size > 0.0:
        if direction == "LONG":
            r_close = (final_exit_price - entry) / risk
        else:
            r_close = (entry - final_exit_price) / risk
        realized_r_accum += active_size * r_close
        if exit_reason == "HORIZON_EXPIRED":
            exit_bars = len(monitored_bars)

    gross_r = realized_r_accum
    net_r = gross_r - cost_in_r
    duration_min = exit_bars * 5
    return gross_r, net_r, exit_reason, duration_min


class Phase11_5HTFEngine:

    def run_research(self) -> Dict[str, Any]:

        t0 = time.perf_counter()

            candles_4h_by_sym[sym] = resample_htf_candles(c5m, bar_ms=4 * 3600 * 1000)
            logger.info("%s: Resampled %d 1H candles, %d 4H candles", sym, len(candles_1h_by_sym[sym]), len(candles_4h_by_sym[sym]))
            
        # 3. Augment each candidate snapshot with HTF features
        for snap in base_snapshots:
            sym = snap.symbol
            c1h = candles_1h_by_sym.get(sym, [])
            c4h = candles_4h_by_sym.get(sym, [])
            htf_feats = compute_htf_features(snap.timestamp_ms, snap.direction, c1h, c4h)

        # 4. Filter strictly for Variant D

    def _cohort_stats(self, net_rs: List[float], gross_rs: List[float], is_wins: List[bool]) -> Dict[str, Any]:
        n = len(net_rs)
        if n == 0:
            return {
                "count": 0, "win_rate": 0.0, "wins": 0, "losses": 0,
                "gross_r": 0.0, "net_r": 0.0, "mean_r": 0.0, "median_r": 0.0, "pf": 0.0,
                "ci_mean_r": [0.0, 0.0], "ci_wr": [0.0, 0.0]
            }
        wins = sum(1 for w in is_wins if w)
        wr = (wins / n) * 100.0
        tot_net = sum(net_rs)
        tot_gross = sum(gross_rs)
        mean_r = float(np.mean(net_rs))
        med_r = float(np.median(net_rs))
        
        g_wins = sum(r for r in gross_rs if r > 0)
        g_losses = abs(sum(r for r in gross_rs if r < 0))
        pf = round(g_wins / max(1e-6, g_losses), 2)
        
        _, ci_l, ci_u = bootstrap_ci(net_rs)
        wr_bool = [1.0 if w else 0.0 for w in is_wins]
        _, wr_l, wr_u = bootstrap_ci(wr_bool)
        
        return {
            "count": n,
            "win_rate": round(wr, 2),
            "wins": wins,
            "losses": n - wins,
            "gross_r": round(tot_gross, 2),
            "net_r": round(tot_net, 2),
            "mean_r": round(mean_r, 3),
            "median_r": round(med_r, 3),
            "pf": pf,
            "ci_mean_r": [round(ci_l, 3), round(ci_u, 3)],
            "ci_wr": [round(wr_l * 100.0, 2), round(wr_u * 100.0, 2)]
        }

    def _audit_htf_regimes(self, var_d_cands: List[HTFCandidateSnapshot]) -> Dict[str, Any]:
        """Evaluates HTF trend alignment, volatility, and structure location on Variant D."""
        test_cands = [s for s in var_d_cands if s.base.split == "TEST"]
        
        def eval_subset(subset: List[HTFCandidateSnapshot]) -> Dict[str, Any]:
            net_rs = [s.base.net_realized_r_base for s in subset]
            gross_rs = [s.base.gross_realized_r for s in subset]
            is_wins = [s.base.exit_reason in ("TP1", "TP2") for s in subset]
            return self._cohort_stats(net_rs, gross_rs, is_wins)
            
        trend_1h_eval = {
            "ALIGNED": eval_subset([s for s in test_cands if s.trend_aligned_1h]),
            "OPPOSING": eval_subset([s for s in test_cands if not s.trend_aligned_1h])
        }
        
        trend_4h_eval = {
            "ALIGNED": eval_subset([s for s in test_cands if s.trend_aligned_4h]),
            "OPPOSING": eval_subset([s for s in test_cands if not s.trend_aligned_4h])
        }
        
        vol_1h_eval = {
            "LOW": eval_subset([s for s in test_cands if s.vol_regime_1h == "LOW"]),
            "NORMAL": eval_subset([s for s in test_cands if s.vol_regime_1h == "NORMAL"]),
            "HIGH": eval_subset([s for s in test_cands if s.vol_regime_1h == "HIGH"])
        }
        
        expansion_1h_eval = {
            "EXPANSION": eval_subset([s for s in test_cands if s.expansion_state_1h == "EXPANSION"]),
            "RANGE": eval_subset([s for s in test_cands if s.expansion_state_1h == "RANGE"])
        }
        
        structure_loc_eval = {
            "FAVORABLE": eval_subset([s for s in test_cands if s.structure_favorable_1h]),
            "UNFAVORABLE": eval_subset([s for s in test_cands if not s.structure_favorable_1h])
        }
        
        combined_htf_eval = {
            "HTF_ALIGNED": eval_subset([s for s in test_cands if s.htf_aligned]),
            "HTF_UNALIGNED": eval_subset([s for s in test_cands if not s.htf_aligned])
        }
        
        return {
            "trend_1h": trend_1h_eval,
            "trend_4h": trend_4h_eval,
            "volatility_1h": vol_1h_eval,
            "expansion_1h": expansion_1h_eval,
            "structure_location_1h": structure_loc_eval,
            "combined_htf_filter": combined_htf_eval
        }

    def _audit_trade_management(self, var_d_cands: List[HTFCandidateSnapshot]) -> Dict[str, Any]:
        """Evaluates 9 discrete trade management policies across Train, Val, and Test."""
        policies = [
            "FIXED_SL_TP",
            "TP1_PARTIAL",
            "BE_0.5R",
            "BE_0.8R",
            "BE_1.0R",
            "STRUCTURAL_TRAILING",
            "ATR_TRAILING",
            "PARTIAL_TRAILING",
            "TIME_EXIT_30M"
        ]
        
        train_cands = [s for s in var_d_cands if s.base.split == "TRAIN"]
        val_cands = [s for s in var_d_cands if s.base.split == "VALIDATION"]
        test_cands = [s for s in var_d_cands if s.base.split == "TEST"]
        
        def run_policy_on_subset(subset: List[HTFCandidateSnapshot], pol: str) -> Dict[str, Any]:
            net_rs = []
            gross_rs = []
            is_wins = []
            for s in subset:
                sym = s.base.symbol
                c5m = self.future_candles_cache.get(sym, [])
                fut = [c for c in c5m if int(c["ts"]) > s.base.timestamp_ms][:24]
                gross_r, net_r, exit_reason, _ = simulate_trade_management(
                    policy_name=pol,
                    direction=s.base.direction,
                    entry=s.base.entry_price,
                    sl=s.base.sl_price,
                    tp1=s.base.tp1_price,
                    tp2=s.base.tp2_price,
                    future_candles_5m=fut,
                    cur_atr=s.base.cur_atr,
                    friction_rate=FRICTION_BASE
                )
                net_rs.append(net_r)
                gross_rs.append(gross_r)
                is_wins.append(gross_r > 0.0)
            return self._cohort_stats(net_rs, gross_rs, is_wins)
            
        policy_eval_matrix = {}
        for pol in policies:
            policy_eval_matrix[pol] = {
                "train": run_policy_on_subset(train_cands, pol),
                "validation": run_policy_on_subset(val_cands, pol),
                "test": run_policy_on_subset(test_cands, pol)
            }
            
        # Select best policy based strictly on Validation Net R / Mean R
        best_policy = max(policies, key=lambda p: policy_eval_matrix[p]["validation"]["mean_r"])
        logger.info("Validation-selected best management policy: %s (Mean R: %.3f)", best_policy, policy_eval_matrix[best_policy]["validation"]["mean_r"])
        
        return {
            "policies_evaluated": policy_eval_matrix,
            "best_policy": best_policy
        }

    def _construct_comparative_matrix(
        self,
        base_snaps: List[CandidateFeatureSnapshot],
        htf_snaps: List[HTFCandidateSnapshot],
        var_d_snaps: List[HTFCandidateSnapshot],
        best_policy: str
    ) -> Dict[str, Any]:
        """Constructs comparative matrix for BASELINE A, VARIANT D, D+HTF, D+MGMT, D+HTF+MGMT."""
        splits = ["TRAIN", "VALIDATION", "TEST"]
        results = {}
        
        combinations = [
            "BASELINE_A",
            "VARIANT_D",
            "VARIANT_D_HTF",
            "VARIANT_D_MANAGEMENT",
            "VARIANT_D_HTF_MANAGEMENT"
        ]
        
        for combo in combinations:
            results[combo] = {}
            for sp in splits:
                if combo == "BASELINE_A":
                    subset = [s for s in base_snaps if s.split == sp]
                    net_rs = [s.net_realized_r_base for s in subset]
                    gross_rs = [s.gross_realized_r for s in subset]
                    is_wins = [s.exit_reason in ("TP1", "TP2") for s in subset]
                elif combo == "VARIANT_D":
                    subset = [s for s in var_d_snaps if s.base.split == sp]
                    net_rs = [s.base.net_realized_r_base for s in subset]
                    gross_rs = [s.base.gross_realized_r for s in subset]
                    is_wins = [s.base.exit_reason in ("TP1", "TP2") for s in subset]
                elif combo == "VARIANT_D_HTF":
                    subset = [s for s in var_d_snaps if s.base.split == sp and s.htf_aligned]
                    net_rs = [s.base.net_realized_r_base for s in subset]
                    gross_rs = [s.base.gross_realized_r for s in subset]
                    is_wins = [s.base.exit_reason in ("TP1", "TP2") for s in subset]
                elif combo == "VARIANT_D_MANAGEMENT":
                    subset = [s for s in var_d_snaps if s.base.split == sp]
                    net_rs, gross_rs, is_wins = [], [], []
                    for s in subset:
                        sym = s.base.symbol
                        fut = [c for c in self.future_candles_cache.get(sym, []) if int(c["ts"]) > s.base.timestamp_ms][:24]
                        gr, nr, _, _ = simulate_trade_management(
                            best_policy, s.base.direction, s.base.entry_price, s.base.sl_price, s.base.tp1_price,
                            s.base.tp2_price, fut, s.base.cur_atr, FRICTION_BASE
                        )
                        net_rs.append(nr)
                        gross_rs.append(gr)
                        is_wins.append(gr > 0.0)
                elif combo == "VARIANT_D_HTF_MANAGEMENT":
                    subset = [s for s in var_d_snaps if s.base.split == sp and s.htf_aligned]
                    net_rs, gross_rs, is_wins = [], [], []
                    for s in subset:
                        sym = s.base.symbol
                        fut = [c for c in self.future_candles_cache.get(sym, []) if int(c["ts"]) > s.base.timestamp_ms][:24]
                        gr, nr, _, _ = simulate_trade_management(
                            best_policy, s.base.direction, s.base.entry_price, s.base.sl_price, s.base.tp1_price,
                            s.base.tp2_price, fut, s.base.cur_atr, FRICTION_BASE
                        )
                        net_rs.append(nr)
                        gross_rs.append(gr)
                        is_wins.append(gr > 0.0)
                        
                results[combo][sp] = self._cohort_stats(net_rs, gross_rs, is_wins)
                
        return results

    def _audit_asset_stability(self, var_d_cands: List[HTFCandidateSnapshot], best_policy: str) -> Dict[str, Any]:
        """Tests whether Variant D + HTF + Management behaves consistently across BTC, ETH, and SOL."""
        test_cands = [s for s in var_d_cands if s.base.split == "TEST" and s.htf_aligned]
        asset_stats = {}
        for sym in self.symbols:
            sub = [s for s in test_cands if s.base.symbol == sym]
            net_rs, gross_rs, is_wins = [], [], []
            for s in sub:
                fut = [c for c in self.future_candles_cache.get(sym, []) if int(c["ts"]) > s.base.timestamp_ms][:24]
                gr, nr, _, _ = simulate_trade_management(
                    best_policy, s.base.direction, s.base.entry_price, s.base.sl_price, s.base.tp1_price,
                    s.base.tp2_price, fut, s.base.cur_atr, FRICTION_BASE
                )
                net_rs.append(nr)
                gross_rs.append(gr)
                is_wins.append(gr > 0.0)
            asset_stats[sym] = self._cohort_stats(net_rs, gross_rs, is_wins)
        return asset_stats

    def test_bootstrap_confidence_interval(self):
        """Verify bootstrap confidence interval computation."""
        sample = [1.0, 1.2, 0.8, -1.0, 1.5, -0.5, 2.0, 1.1]
        mean, lower, upper = bootstrap_ci(sample, n_iterations=500)
        self.assertAlmostEqual(mean, float(np.mean(sample)), places=4)
        self.assertLess(lower, mean)
        self.assertGreater(upper, mean)

    def test_multiple_testing_benjamini_hochberg(self):
        """Verify Benjamini-Hochberg FDR ranking and critical values."""
        p_vals = [0.005, 0.02, 0.04, 0.15, 0.30]
        m = len(p_vals)
        ranked = []
        for rank, p in enumerate(p_vals, 1):
            crit = (rank / m) * 0.05
            survives = p <= crit
            ranked.append((p, crit, survives))

        # Rank 1: 0.005 <= (1/5)*0.05 = 0.010 -> True
        self.assertTrue(ranked[0][2])
        # Rank 2: 0.020 <= (2/5)*0.05 = 0.020 -> True
        self.assertTrue(ranked[1][2])
        # Rank 3: 0.040 <= (3/5)*0.05 = 0.030 -> False
        self.assertFalse(ranked[2][2])
