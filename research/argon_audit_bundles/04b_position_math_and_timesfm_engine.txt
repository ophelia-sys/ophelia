"""Authoritative Position Accounting, Mathematical Breakeven, and Dynamic Trailing Logic for Ophelia.

Layer: control_plane/position_math
Invariants:
1. Actual Fill Accounting: No calculations from requested order parameters.
   Fills govern entry price, filled qty, remaining qty, fees, and realized PnL.
2. Mathematically Correct Net Breakeven:
   Stop is NOT raw entry price.
   Calculates the minimum protective price representing the true cost basis
   of the remaining position, incorporating entry fees, TP1 exit fees,
   remaining exit fees, and funding.
3. Monotonic Dynamic Trailing:
   For LONG: new_stop >= previous_stop (never moves down).
   For SHORT: new_stop <= previous_stop (never moves up).
4. Deterministic Tick Precision:
   Stops are rounded to instrument tick size (away from danger).
"""
from __future__ import annotations

from dataclasses import dataclass, asdict
import math
import time
from typing import Any, Dict, Optional


@dataclass(frozen=True)
class FeeModel:
    """Authoritative fee configuration governance model.
    
    Decouples fee accounting from hidden constants, enabling explicit
    auditability across SHADOW, PAPER, and LIVE execution scopes.
    """
    fee_source: str             # e.g. "SHADOW_CONFIG", "BINGX_STANDARD", "BINGX_VIP", "ACCOUNT_OVERRIDE"
    fee_type: str               # "PERCENTAGE", "FIXED", "TIERED"
    maker_fee: float            # e.g. 0.0002 (0.02%)
    taker_fee: float            # e.g. 0.0005 (0.05%)
    effective_at: int = 0       # timestamp_ms
    instrument_scope: str = "ALL"  # "ALL", "CRYPTO", or specific symbol "BTC-USDT"
    account_scope: str = "ALL"     # "ALL" or specific account_id
    mode: str = "SHADOW"           # "SHADOW", "PAPER", "LIVE"

    def get_taker_fee_rate(self) -> float:
        return self.taker_fee

    def get_maker_fee_rate(self) -> float:
        return self.maker_fee

    def to_dict(self) -> Dict[str, Any]:
        return asdict(self)

def register_fee_model(model: FeeModel) -> None:
    """Register or override a fee model for a specific mode or instrument scope."""
    key = f"{model.mode.upper()}:{model.instrument_scope.upper()}:{str(model.account_scope).upper()}"
    _active_fee_models[key] = model
    _active_fee_models[model.mode.upper()] = model


def resolve_fee_model(mode: str = "SHADOW", symbol: str = "ALL", account_id: Any = "ALL") -> FeeModel:
    """Resolve the authoritative configured fee model for the given mode, symbol, and account."""
    m = (mode or "SHADOW").upper()
    sym = (symbol or "ALL").upper()
    acc = str(account_id or "ALL").upper()

    k1 = f"{m}:{sym}:{acc}"
    if k1 in _active_fee_models:
        return _active_fee_models[k1]

    k2 = f"{m}:{sym}:ALL"
    if k2 in _active_fee_models:
        return _active_fee_models[k2]

    k3 = f"{m}:ALL:ALL"
    if k3 in _active_fee_models:
        return _active_fee_models[k3]

    if m in _active_fee_models:
        return _active_fee_models[m]

    return SHADOW_FEE_MODEL

def round_to_tick(price: float, tick_size: float, direction: str = "LONG") -> float:
    """Round price to the nearest tick size, protecting the stop (away from danger)."""
    if tick_size <= 0:
        return price
    ticks = price / tick_size
    if direction.upper() == "LONG":
        return round(math.ceil(ticks) * tick_size, 8)
    else:
        return round(math.floor(ticks) * tick_size, 8)


def get_instrument_precision(symbol: str) -> Dict[str, float]:
    """Return tick size and quantity precision for an instrument."""
    s = symbol.upper()
    if "BTC" in s:
        return {"tick_size": 0.1, "qty_precision": 4, "price_precision": 1}
    elif "ETH" in s:
        return {"tick_size": 0.01, "qty_precision": 3, "price_precision": 2}
    elif "SOL" in s:
        return {"tick_size": 0.01, "qty_precision": 2, "price_precision": 2}
    elif "XRP" in s:
        return {"tick_size": 0.0001, "qty_precision": 1, "price_precision": 4}
    elif "GOLD" in s or "XAU" in s:
        return {"tick_size": 0.01, "qty_precision": 2, "price_precision": 2}
    else:
        return {"tick_size": 0.01, "qty_precision": 2, "price_precision": 2}


def calculate_actual_fee(
    price: float,
    qty: float,
    fee_rate: Optional[float] = None,
    fee_model: Optional[FeeModel] = None
) -> float:
    """Calculate actual transaction fee for a fill using explicit FeeModel or rate."""
    if price <= 0 or qty <= 0:
        return 0.0
    if fee_rate is not None:
        rate = fee_rate
    elif fee_model is not None:
        rate = fee_model.get_taker_fee_rate()
    else:
        rate = resolve_fee_model().get_taker_fee_rate()
    return round(price * qty * rate, 8)


def calculate_realized_pnl(
    entry_price: float,
    exit_price: float,
    qty: float,
    direction: str,
    fee: float = 0.0
) -> float:
    """Calculate net realized PnL for a closed portion of a position."""
    if entry_price <= 0 or exit_price <= 0 or qty <= 0:
        return 0.0
    is_long = direction.upper() == "LONG"
    gross_pnl = (exit_price - entry_price) * qty * (1.0 if is_long else -1.0)
    return round(gross_pnl - fee, 6)


def calculate_unrealized_pnl(
    entry_price: float,
    current_price: float,
    qty: float,
    direction: str
) -> float:
    """Calculate gross unrealized PnL based on mark/current price."""
    if entry_price <= 0 or current_price <= 0 or qty <= 0:
        return 0.0
    is_long = direction.upper() == "LONG"
    return round((current_price - entry_price) * qty * (1.0 if is_long else -1.0), 6)


def calculate_net_breakeven_stop(
    entry_price: float,
    initial_qty: float,
    remaining_qty: float,
    direction: str,
    entry_fee: float,
    tp1_realized_pnl: float,
    tp1_fee: float,
    funding_accrued: float = 0.0,
    fee_rate: Optional[float] = None,
    fee_model: Optional[FeeModel] = None,
    tick_size: float = 0.01,
    previous_stop: Optional[float] = None
) -> Dict[str, Any]:
    """Calculate the mathematically correct net breakeven stop for the remaining position.
    
    Invariants:
    1. NEVER raw entry price: accounts for entry fee, partial exit fee, remaining exit fee, funding.
    2. Cost basis of remaining leg:
       LONG:  entry * (1 + fee_rate) / (1 - fee_rate)
       SHORT: entry * (1 - fee_rate) / (1 + fee_rate)
    3. Net trade breakeven: Price at which total trade Net PnL (TP1 profit + remaining PnL - all fees) >= 0.
    4. Monotonicity:
       LONG:  breakeven_stop >= previous_stop
       SHORT: breakeven_stop <= previous_stop
    5. Tick rounding:
       LONG:  ceil to next tick (away from danger)
       SHORT: floor to next tick (away from danger)
    """
    is_long = direction.upper() == "LONG"
    now_ms = int(time.time() * 1000)

    # Resolve effective fee model and rate
    if fee_model is not None:
        effective_fee_model = fee_model
        effective_rate = fee_model.get_taker_fee_rate() if fee_rate is None else fee_rate
    elif fee_rate is not None:
        effective_fee_model = resolve_fee_model()
        effective_rate = fee_rate
    else:
        effective_fee_model = resolve_fee_model()
        effective_rate = effective_fee_model.get_taker_fee_rate()

    if remaining_qty <= 0 or entry_price <= 0:
        return {
            "breakeven_stop": previous_stop or entry_price,
            "calculated_cost_basis": entry_price,
            "calculation_inputs": {},
            "calculated_at_ms": now_ms
        }

    # 1. Cost basis of remaining position covering its roundtrip entry & exit fees
    if is_long:
        cost_basis = entry_price * ((1.0 + effective_rate) / (1.0 - effective_rate))
    else:
        cost_basis = entry_price * ((1.0 - effective_rate) / (1.0 + effective_rate))

    # 2. Total trade breakeven price (accounting for TP1 profit and all fees)
    prior_net_cash = tp1_realized_pnl - (entry_fee + tp1_fee + funding_accrued)
    if is_long:
        # (P_be - P_entry) * remaining_qty - P_be * remaining_qty * effective_rate + prior_net_cash >= 0
        denom = remaining_qty * (1.0 - effective_rate)
        if denom > 0:
            total_be_price = (entry_price * remaining_qty - prior_net_cash) / denom
        else:
            total_be_price = cost_basis
        # Protect at least cost basis to guarantee non-negative remaining leg
        candidate_stop = max(cost_basis, total_be_price)
        # Round UP to tick size (conservative)
        if tick_size > 0:
            candidate_stop = math.ceil(candidate_stop / tick_size) * tick_size
        candidate_stop = round(candidate_stop, 6)
        # Monotonicity: stop only moves UP for LONG
        if previous_stop is not None and previous_stop > 0:
            final_stop = max(previous_stop, candidate_stop)
        else:
            final_stop = candidate_stop
    else:
        # (P_entry - P_be) * remaining_qty - P_be * remaining_qty * effective_rate + prior_net_cash >= 0
        denom = remaining_qty * (1.0 + effective_rate)
        if denom > 0:
            total_be_price = (entry_price * remaining_qty + prior_net_cash) / denom
        else:
            total_be_price = cost_basis
        # Protect at least cost basis
        candidate_stop = min(cost_basis, total_be_price)
        # Round DOWN to tick size (conservative)
        if tick_size > 0:
            candidate_stop = math.floor(candidate_stop / tick_size) * tick_size
        candidate_stop = round(candidate_stop, 6)
        # Monotonicity: stop only moves DOWN for SHORT
        if previous_stop is not None and previous_stop > 0:
            final_stop = min(previous_stop, candidate_stop)
        else:
            final_stop = candidate_stop

    inputs = {
        "entry_price": entry_price,
        "initial_qty": initial_qty,
        "remaining_qty": remaining_qty,
        "direction": direction.upper(),
        "entry_fee": entry_fee,
        "tp1_realized_pnl": tp1_realized_pnl,
        "tp1_fee": tp1_fee,
        "funding_accrued": funding_accrued,
        "fee_rate": effective_rate,
        "fee_model": effective_fee_model.to_dict() if effective_fee_model else None,
        "fee_source": effective_fee_model.fee_source if effective_fee_model else "CUSTOM",
        "tick_size": tick_size,
        "previous_stop": previous_stop,
        "cost_basis_raw": round(cost_basis, 6),
        "total_be_raw": round(total_be_price, 6)
    }

    return {
        "breakeven_stop": final_stop,
        "calculated_cost_basis": round(cost_basis, 6),
        "calculation_inputs": inputs,
        "calculated_at_ms": now_ms
    }


def calculate_dynamic_trailing_stop(
    current_price: float,
    entry_price: float,
    direction: str,
    current_stop: float,
    trailing_ref: float,
    atr: float = 0.0,
    volatility: float = 0.0,
    regime: str = "TREND",
    trailing_pct: float = 1.5,
    tick_size: float = 0.01,
    min_move_pct: float = 0.001
) -> Dict[str, Any]:
    """Calculate server-side dynamic trailing stop with strict monotonicity.
    
    Invariants:
    1. LONG: new_stop >= current_stop (monotonic non-decreasing).
    2. SHORT: new_stop <= current_stop (monotonic non-increasing).
    3. Trailing method incorporates ATR & volatility regime if available, else trailing_pct.
    4. Threshold gating: updates emitted only on meaningful price progress (>= min_move_pct or 2 ticks).
    """
    is_long = direction.upper() == "LONG"
    now_ms = int(time.time() * 1000)

    if current_price <= 0 or entry_price <= 0:
        return {
            "new_stop": current_stop,
            "ref_price": trailing_ref or entry_price,
            "moved": False,
            "reason": "INVALID_PRICE"
        }

    # Reference peak price (high-water mark for LONG, low-water mark for SHORT)
    if is_long:
        ref_price = max(trailing_ref or entry_price, current_price)
    else:
        ref_price = min(trailing_ref or entry_price, current_price) if (trailing_ref and trailing_ref > 0) else current_price

    # Trailing distance calculation
    if atr > 0:
        mult = 2.0 if regime == "TREND" else (1.5 if regime in ("HIGH", "VOLATILE") else 2.5)
        trail_distance = max(mult * atr, ref_price * (trailing_pct / 100.0))
        method = "ATR_DYNAMIC"
    else:
        trail_distance = ref_price * (trailing_pct / 100.0)
        method = "PERCENTAGE_TRAILING"

    # Candidate calculation
    if is_long:
        raw_stop = ref_price - trail_distance
        if tick_size > 0:
            candidate_stop = math.floor(raw_stop / tick_size) * tick_size
        else:
            candidate_stop = raw_stop
        candidate_stop = round(candidate_stop, 6)
        # Monotonicity clamp
        improved_stop = max(current_stop or 0.0, candidate_stop)
        stop_advanced = improved_stop > (current_stop or 0.0)
    else:
        raw_stop = ref_price + trail_distance
        if tick_size > 0:
            candidate_stop = math.ceil(raw_stop / tick_size) * tick_size
        else:
            candidate_stop = raw_stop
        candidate_stop = round(candidate_stop, 6)
        # Monotonicity clamp
        improved_stop = min(current_stop or 999999.0, candidate_stop) if (current_stop and current_stop > 0) else candidate_stop
        stop_advanced = improved_stop < (current_stop or 999999.0)

    # Meaningful movement threshold check
    threshold = max(2.0 * tick_size, (current_stop or entry_price) * min_move_pct)
    meaningful = abs(improved_stop - (current_stop or 0.0)) >= threshold

    profit_distance = (current_price - entry_price) * (1.0 if is_long else -1.0)

    trailing_inputs = {
        "current_price": current_price,
        "entry_price": entry_price,
        "direction": direction.upper(),
        "ref_price": ref_price,
        "atr": atr,
        "volatility": volatility,
        "regime": regime,
        "trailing_pct": trailing_pct,
        "trail_distance": round(trail_distance, 6),
        "tick_size": tick_size,
        "profit_distance": round(profit_distance, 6),
        "method": method
    }

    return {
        "new_stop": improved_stop,
        "ref_price": ref_price,
        "moved": stop_advanced and meaningful,
        "trailing_method": method,
        "trailing_inputs": trailing_inputs,
        "last_update_ms": now_ms,
        "reason": f"TRAILING_ADVANCED_{method}" if stop_advanced else "NO_ADVANCEMENT",
        "profit_at_update": round(profit_distance, 4),
        "regime_at_update": regime
    }

        hist_candles = [
            {"ts": 1000 + i * 300000, "open": 100.0, "high": 102.0, "low": 99.0, "close": 101.0, "volume": 500.0}
            for i in range(50)

        cand_ts = 1000 + 49 * 300000

        # Causal slice strictly <= cand_ts
        slice_orig = [c for c in hist_candles if c["ts"] <= cand_ts]

        # Corrupted future bar at cand_ts + 300000
        corrupted_future = copy.deepcopy(hist_candles)
        corrupted_future.append({"ts": cand_ts + 300000, "open": 9999.0, "high": 9999.0, "low": 9999.0, "close": 9999.0, "volume": 999999.0})
        slice_after_corruption = [c for c in corrupted_future if c["ts"] <= cand_ts]

        self.assertEqual(len(slice_orig), len(slice_after_corruption))
        self.assertEqual(slice_orig[-1]["close"], slice_after_corruption[-1]["close"])

    def test_multiple_testing_fdr_logic(self):
        """Verifies Benjamini-Hochberg FDR ranking and critical threshold logic."""
        raw_p = [0.0001, 0.002, 0.045, 0.35]
        m = len(raw_p)
        alpha = 0.05
        # Expected BH critical thresholds:
        # Rank 1: (1/4)*0.05 = 0.0125 -> 0.0001 <= 0.0125 (REJECT NULL)
        # Rank 2: (2/4)*0.05 = 0.0250 -> 0.0020 <= 0.0250 (REJECT NULL)
        # Rank 3: (3/4)*0.05 = 0.0375 -> 0.0450 > 0.0375 (FAIL TO REJECT)
        # Rank 4: (4/4)*0.05 = 0.0500 -> 0.3500 > 0.0500 (FAIL TO REJECT)
        crits = [(i + 1) / m * alpha for i in range(m)]
        statuses = ["REJECT_NULL" if raw_p[i] <= crits[i] else "FAIL_TO_REJECT" for i in range(m)]
        self.assertEqual(statuses, ["REJECT_NULL", "REJECT_NULL", "FAIL_TO_REJECT", "FAIL_TO_REJECT"])

def evaluate_forward_trajectory_phase12(
    cand: Dict[str, Any],
    future_bars: List[dict]
) -> Tuple[str, float, int, float, float, float, float, float, float, float, float, float]:
    """Causal trace over future closed 5m bars.
    SL evaluated strictly before TP on same candle to eliminate favorable bias.
    """
    direction = cand["direction"].upper()
    entry = float(cand["entry_price"])
    sl = float(cand["sl_price"])
    tp1 = float(cand["tp1_price"])

    risk = (entry - sl) if direction == "LONG" else (sl - entry)
    risk = max(1e-6, risk)

    cost_r_zero = 0.0
    cost_r_low = (FRICTION_LOW * entry) / risk
    cost_r_base = (FRICTION_BASE * entry) / risk
    cost_r_high = (FRICTION_HIGH * entry) / risk

    if not future_bars:
        return ("NO_DATA", entry, 0, 0.0, 0.0, -cost_r_low, -cost_r_base, -cost_r_high, 0.0, 0.0, entry, entry)

    highs = [float(b["high"]) for b in future_bars]
    lows = [float(b["low"]) for b in future_bars]
    closes = [float(b["close"]) for b in future_bars]

    actual_low = min(lows)
    actual_high = max(highs)

    exit_reason = "HORIZON_EXPIRED"
    time_to_exit_min = len(future_bars) * 5
    exit_price = closes[-1]

    if direction == "LONG":
        mfe = actual_high - entry
        mae = actual_low - entry
        for idx, b in enumerate(future_bars):
            bh, bl = float(b["high"]), float(b["low"])
            if bl <= sl:
                exit_reason = "SL"
                time_to_exit_min = (idx + 1) * 5
                exit_price = sl
                break
            elif bh >= tp1:
                exit_reason = "TP1"
                time_to_exit_min = (idx + 1) * 5
                exit_price = tp1
                break
        gross_r = (exit_price - entry) / risk
    else:  # SHORT
        mfe = entry - actual_low
        mae = entry - actual_high
        for idx, b in enumerate(future_bars):
            bh, bl = float(b["high"]), float(b["low"])
            if bh >= sl:
                exit_reason = "SL"
                time_to_exit_min = (idx + 1) * 5
                exit_price = sl
                break
            elif bl <= tp1:
                exit_reason = "TP1"
                time_to_exit_min = (idx + 1) * 5
                exit_price = tp1
                break
        gross_r = (entry - exit_price) / risk

    net_r_zero = gross_r
    net_r_low = gross_r - cost_r_low
    net_r_base = gross_r - cost_r_base
    net_r_high = gross_r - cost_r_high

    return (exit_reason, exit_price, time_to_exit_min, gross_r,
            net_r_zero, net_r_low, net_r_base, net_r_high,
            mfe, mae, actual_low, actual_high)

    def _compute_metrics(self) -> Dict[str, Any]:
        """Calculates authoritative quantitative, statistical, economic, and multiple testing metrics."""
        test_cands = [c for c in self.all_candidates if c.split == "TEST"]
        val_cands = [c for c in self.all_candidates if c.split == "VALIDATION"]
        train_cands = [c for c in self.all_candidates if c.split == "TRAIN"]

        def cohort_stats(subset: List[Phase12Candidate], friction_key: str = "base", use_reforecast: bool = False) -> Dict[str, Any]:
            if not subset:
                return {
                    "count": 0, "win_rate": 0.0, "tp1_count": 0, "sl_count": 0,
                    "gross_r": 0.0, "net_r": 0.0, "mean_r": 0.0, "median_r": 0.0,
                    "pf": 0.0, "max_drawdown": 0.0, "ci_mean_r": [0.0, 0.0], "ci_wr": [0.0, 0.0]
                }
            n = len(subset)
            wins = sum(1 for c in subset if c.exit_reason in ("TP1", "TP2"))
            sl_count = sum(1 for c in subset if c.exit_reason == "SL")
            wr = (wins / n) * 100.0

            if use_reforecast:
                net_rs = [c.reforecast_net_r_base for c in subset]
            elif friction_key == "zero":
                net_rs = [c.net_realized_r_zero for c in subset]
            elif friction_key == "low":
                net_rs = [c.net_realized_r_low for c in subset]
            elif friction_key == "high":
                net_rs = [c.net_realized_r_high for c in subset]
            else:
                net_rs = [c.net_realized_r_base for c in subset]

            gross_rs = [c.gross_realized_r for c in subset]
            tot_gross = sum(gross_rs)
            tot_net = sum(net_rs)
            mean_r = float(np.mean(net_rs))
            median_r = float(np.median(net_rs))

            gw = sum(r for r in gross_rs if r > 0)
            gl = abs(sum(r for r in gross_rs if r < 0))
            pf = round(gw / max(1e-6, gl), 2)

            # Max Drawdown in R
            cum_r = np.cumsum(net_rs)
            peaks = np.maximum.accumulate(cum_r)
            drawdowns = peaks - cum_r
            max_dd = float(np.max(drawdowns)) if len(drawdowns) > 0 else 0.0

            # 95% Bootstrap CI
            _, ci_l, ci_u = bootstrap_ci(net_rs)
            wr_bool = [1.0 if c.exit_reason in ("TP1", "TP2") else 0.0 for c in subset]
            _, wr_l, wr_u = bootstrap_ci(wr_bool)

            return {
                "count": n,
                "win_rate": round(wr, 2),
                "tp1_count": wins,
                "sl_count": sl_count,
                "gross_r": round(tot_gross, 2),
                "net_r": round(tot_net, 2),
                "mean_r": round(mean_r, 3),
                "median_r": round(median_r, 3),
                "pf": pf,
                "max_drawdown": round(max_dd, 2),
                "ci_mean_r": [round(ci_l, 3), round(ci_u, 3)],
                "ci_wr": [round(wr_l * 100.0, 2), round(wr_u * 100.0, 2)]
            }

        # 1. Baseline Progression Cohorts Definition:
        # A = BASELINE_A (All candidates)
        # B = VARIANT_D (Passes Variant D structural quality)
        # C = D4H (Variant D + 4H trend alignment)
        # D = D4H + TimesFM Reachability (tfm_reachable_binary_60m == True)
        # E = D + Downside Gate (pass_downside_gate == True)
        # F = E + Post-Entry Reforecast Management
        def filter_cohort(subset: List[Phase12Candidate], cohort_key: str) -> List[Phase12Candidate]:
            if cohort_key == "A_BASELINE":
                return subset
            elif cohort_key == "B_VARIANT_D":
                return [c for c in subset if c.pass_variant_d]
            elif cohort_key == "C_D4H":
                return [c for c in subset if c.pass_d4h]
            elif cohort_key == "D_D4H_TFM_REACHABLE":
                return [c for c in subset if c.pass_d4h and c.tfm_reachable_binary_60m]
            elif cohort_key == "E_D4H_TFM_DOWNSIDE":
                return [c for c in subset if c.pass_d4h and c.tfm_reachable_binary_60m and c.pass_downside_gate]
            elif cohort_key == "F_D4H_TFM_MANAGED":
                return [c for c in subset if c.pass_d4h and c.tfm_reachable_binary_60m and c.pass_downside_gate]
            return []

        cohort_keys = [
            "A_BASELINE", "B_VARIANT_D", "C_D4H",
            "D_D4H_TFM_REACHABLE", "E_D4H_TFM_DOWNSIDE", "F_D4H_TFM_MANAGED"
        ]

        matrix_results = {}
        for k in cohort_keys:
            use_ref = (k == "F_D4H_TFM_MANAGED")
            matrix_results[k] = {
                "train": cohort_stats(filter_cohort(train_cands, k), use_reforecast=use_ref),
                "validation": cohort_stats(filter_cohort(val_cands, k), use_reforecast=use_ref),
                "test": cohort_stats(filter_cohort(test_cands, k), use_reforecast=use_ref)
            }

        # 2. Incremental Deltas on Untouched TEST Set:
        test_a = matrix_results["A_BASELINE"]["test"]
        test_b = matrix_results["B_VARIANT_D"]["test"]
        test_c = matrix_results["C_D4H"]["test"]
        test_d = matrix_results["D_D4H_TFM_REACHABLE"]["test"]
        test_e = matrix_results["E_D4H_TFM_DOWNSIDE"]["test"]
        test_f = matrix_results["F_D4H_TFM_MANAGED"]["test"]

        incremental_deltas_test = {
            "B_minus_A_structure_quality": {

                "delta_net_r": round(test_b["net_r"] - test_a["net_r"], 2),
                "delta_mean_r": round(test_b["mean_r"] - test_a["mean_r"], 3),
                "delta_pf": round(test_b["pf"] - test_a["pf"], 2)
            },
            "C_minus_B_4h_alignment": {
                "delta_n": test_c["count"] - test_b["count"],
                "delta_win_rate_pct": round(test_c["win_rate"] - test_b["win_rate"], 2),
                "delta_gross_r": round(test_c["gross_r"] - test_b["gross_r"], 2),
                "delta_net_r": round(test_c["net_r"] - test_b["net_r"], 2),
                "delta_mean_r": round(test_c["mean_r"] - test_b["mean_r"], 3),
                "delta_pf": round(test_c["pf"] - test_b["pf"], 2)
            },
            "D_minus_C_timesfm_reachability": {
                "delta_n": test_d["count"] - test_c["count"],
                "delta_win_rate_pct": round(test_d["win_rate"] - test_c["win_rate"], 2),
                "delta_gross_r": round(test_d["gross_r"] - test_c["gross_r"], 2),
                "delta_net_r": round(test_d["net_r"] - test_c["net_r"], 2),
                "delta_mean_r": round(test_d["mean_r"] - test_c["mean_r"], 3),
                "delta_pf": round(test_d["pf"] - test_c["pf"], 2)
            },
            "E_minus_D_timesfm_downside_gate": {
                "delta_n": test_e["count"] - test_d["count"],
                "delta_win_rate_pct": round(test_e["win_rate"] - test_d["win_rate"], 2),
                "delta_gross_r": round(test_e["gross_r"] - test_d["gross_r"], 2),
                "delta_net_r": round(test_e["net_r"] - test_d["net_r"], 2),
                "delta_mean_r": round(test_e["mean_r"] - test_d["mean_r"], 3),
                "delta_pf": round(test_e["pf"] - test_d["pf"], 2)
            },
            "F_minus_E_reforecast_management": {
                "delta_n": test_f["count"] - test_e["count"],
                "delta_win_rate_pct": round(test_f["win_rate"] - test_e["win_rate"], 2),
                "delta_gross_r": round(test_f["gross_r"] - test_e["gross_r"], 2),
                "delta_net_r": round(test_f["net_r"] - test_e["net_r"], 2),
                "delta_mean_r": round(test_f["mean_r"] - test_e["mean_r"], 3),
                "delta_pf": round(test_f["pf"] - test_e["pf"], 2)
            }
        }

        # 3. TimesFM Horizon Sensitivity on D4H (TEST split)
        d4h_test = filter_cohort(test_cands, "C_D4H")
        horizon_reachability_test = {
            "h15m": cohort_stats([c for c in d4h_test if c.tfm_reachability_ratio_15m >= 1.0]),
            "h30m": cohort_stats([c for c in d4h_test if c.tfm_reachability_ratio_30m >= 1.0]),
            "h45m": cohort_stats([c for c in d4h_test if c.tfm_reachability_ratio_45m >= 1.0]),
            "h60m_primary": cohort_stats([c for c in d4h_test if c.tfm_reachability_ratio_60m >= 1.0])
        }

        # 4. TimesFM vs Simple Baselines (Point Prediction MSE on TEST split)
        mse_tfm = float(np.mean([(c.tfm_p60 - c.exit_price)**2 for c in test_cands]))
        mse_persist = float(np.mean([(c.base_persistence_pred - c.exit_price)**2 for c in test_cands]))
        mse_ema = float(np.mean([(c.base_ema_extrap_pred - c.exit_price)**2 for c in test_cands]))
        mse_vwap = float(np.mean([(c.base_vwap_pred - c.exit_price)**2 for c in test_cands]))
        mse_ar = float(np.mean([(c.base_ar_pred - c.exit_price)**2 for c in test_cands]))

        baseline_comparison = {
            "timesfm_3_0_rmse": round(math.sqrt(mse_tfm), 2),
            "persistence_rmse": round(math.sqrt(mse_persist), 2),
            "ema_extrapolation_rmse": round(math.sqrt(mse_ema), 2),
            "session_vwap_rmse": round(math.sqrt(mse_vwap), 2),
            "statistical_ar_rmse": round(math.sqrt(mse_ar), 2)
        }

        # 5. Downside / Stop-Margin Analysis (TEST split)
        mae_coverage_pct = round(sum(1 for c in d4h_test if c.tfm_q10_covered_mae) / max(1, len(d4h_test)) * 100.0, 2)
        mae_errors = [c.tfm_mae_prediction_error for c in d4h_test]
        sl_touch_prec = round(sum(1 for c in d4h_test if c.tfm_sl_touch_predicted and c.exit_reason == "SL") /
                              max(1, sum(1 for c in d4h_test if c.tfm_sl_touch_predicted)) * 100.0, 2)

        downside_analysis = {
            "q10_coverage_mae_pct": mae_coverage_pct,
            "mae_prediction_error_mean": round(float(np.mean(mae_errors)), 2) if mae_errors else 0.0,
            "mae_prediction_rmse": round(float(np.sqrt(np.mean(np.array(mae_errors)**2))), 2) if mae_errors else 0.0,
            "sl_touch_prediction_precision_pct": sl_touch_prec
        }

        # 6. Asset Breakdown (TEST split for D4H and D4H + TimesFM)
        asset_breakdown = {}
        for sym in self.symbols:
            sym_d4h = [c for c in d4h_test if c.symbol == sym]

                "d4h_timesfm": cohort_stats(sym_tfm),
                "delta_net_r": round(cohort_stats(sym_tfm)["net_r"] - cohort_stats(sym_d4h)["net_r"], 2),

        regimes = ["TREND_EXPANSION", "RANGE_CHOP"]

        tfm_test = [c for c in d4h_test if c.tfm_reachable_binary_60m]

        # 9. Clustering & Effective Sample Size (TEST split)
        day_map: Dict[str, List[Phase12Candidate]] = {}
        for c in test_cands:

        cluster_sizes = [len(v) for v in day_map.values()]

            "mean_cluster_size": round(mean_c_size, 2),
            "max_cluster_size": max(cluster_sizes) if cluster_sizes else 0,
            "effective_sample_size_n_eff": round(n_eff, 1)

        # 10. Multiple-Testing Registry & Benjamini-Hochberg FDR
        # Hypotheses:
        # H1: Target Reachability improves Net Mean R
        # H2: Downside Distribution provides informative MAE/SL protection
        # H3: Target Distribution differentiates feasible vs infeasible targets
        # H4: T+15m Reforecast improves post-entry net expectancy
        # Calculate two-sided p-value via Welch's t-test approximation / bootstrap
        r_d4h = [c.net_realized_r_base for c in d4h_test]
        r_tfm = [c.net_realized_r_base for c in tfm_test]

        from scipy import stats as sp_stats
        t_stat, p_val_h1 = sp_stats.ttest_ind(r_tfm, r_d4h, equal_var=False) if (len(r_tfm) > 2 and len(r_d4h) > 2) else (0.0, 1.0)

        # H2 p-value: correlation between predicted drawdown and actual MAE
        pred_dd = [c.tfm_predicted_drawdown for c in d4h_test]
        act_mae = [c.mae for c in d4h_test]
        r_dd, p_val_h2 = sp_stats.pearsonr(pred_dd, act_mae) if len(d4h_test) > 5 else (0.0, 1.0)

        # H3 p-value: Difference in TP1 hit rate between reachable and unreachable

        wr_reach = [1.0 if c.exit_reason in ("TP1", "TP2") else 0.0 for c in tfm_test]
        wr_unreach = [1.0 if c.exit_reason in ("TP1", "TP2") else 0.0 for c in unreach_d4h]
        _, p_val_h3 = sp_stats.ttest_ind(wr_reach, wr_unreach, equal_var=False) if (len(wr_reach) > 2 and len(wr_unreach) > 2) else (0.0, 1.0)
        p_val_h3 = float(p_val_h3)

        # H4 p-value: Managed vs Unmanaged Net R on Cohort E
        cohort_e = filter_cohort(test_cands, "E_D4H_TFM_DOWNSIDE")
        r_unmanaged = [c.net_realized_r_base for c in cohort_e]
        r_managed = [c.reforecast_net_r_base for c in cohort_e]
        _, p_val_h4 = sp_stats.ttest_rel(r_managed, r_unmanaged) if len(cohort_e) > 5 else (0.0, 1.0)
        p_val_h4 = float(p_val_h4)

        raw_hypotheses = [
            {"id": "H1", "name": "Target Reachability Improves Net Mean R", "p_value": p_val_h1},
            {"id": "H2", "name": "Downside Distribution Informs MAE/SL", "p_value": p_val_h2},
            {"id": "H3", "name": "Distribution Differentiates Feasible Targets", "p_value": p_val_h3},
            {"id": "H4", "name": "T+15m Reforecast Improves Trade Management", "p_value": p_val_h4}
        ]
        raw_hypotheses.sort(key=lambda x: x["p_value"])
        m = len(raw_hypotheses)
        fdr_alpha = 0.05
        multiple_testing_registry = []
        for i, h in enumerate(raw_hypotheses):
            rank = i + 1
            crit = (rank / m) * fdr_alpha
            bonf = fdr_alpha / m
            p_val = h["p_value"]
            status = "REJECT_NULL (SUPPORTED)" if p_val <= crit else "FAIL_TO_REJECT (NOT SUPPORTED)"
            multiple_testing_registry.append({
                "rank": rank,
                "hypothesis_id": h["id"],
                "name": h["name"],
                "raw_p_value": round(p_val, 5),
                "bh_critical_value": round(crit, 5),
                "bonferroni_critical": round(bonf, 5),
                "fdr_status": status
            })

        # 11. Final Stop Condition Evaluation
        net_r_tfm = test_d["net_r"]
        mean_r_tfm = test_d["mean_r"]
        delta_net_r = incremental_deltas_test["D_minus_C_timesfm_reachability"]["delta_net_r"]
        delta_mean_r = incremental_deltas_test["D_minus_C_timesfm_reachability"]["delta_mean_r"]

        if net_r_tfm > 0 and delta_net_r > 0 and p_val_h1 <= 0.05:
            scientific_conclusion = "OUT-OF-SAMPLE SUPPORTED: TimesFM provides statistically significant incremental economic value."
            stop_condition_met = "PROSPECTIVE_SHADOW_CANDIDATE"
        elif delta_net_r > 0 and delta_mean_r > 0:
            scientific_conclusion = "PROMISING / RESEARCH HYPOTHESIS: TimesFM improves net metrics incrementally but does not achieve statistically significant positive-net profitability."
            stop_condition_met = "TIMESFM INCREMENTAL VALUE PARTIAL / NOT FULLY ESTABLISHED"
        else:
            scientific_conclusion = "OUT-OF-SAMPLE NOT SUPPORTED: TimesFM target reachability does not improve net expectancy on fresh chronological test data."
            stop_condition_met = "TIMESFM INCREMENTAL VALUE NOT ESTABLISHED"

        return {
            "spec_hash": D4H_SPEC_HASH,
            "specification": D4H_SPECIFICATION,
            "dataset_period": "2023-12-01 to 2024-03-31 (122 calendar days)",
            "symbols": self.symbols,
            "candidate_counts": {
                "total": len(self.all_candidates),
                "train": len(train_cands),
                "validation": len(val_cands),
                "test": len(test_cands),
                "d4h_test": len(d4h_test),
                "tfm_reachable_test": len(tfm_test)
            },
            "baseline_progression_matrix": matrix_results,
            "incremental_deltas_test": incremental_deltas_test,
            "horizon_sensitivity_test": horizon_reachability_test,
            "baseline_models_comparison": baseline_comparison,
            "downside_distribution_analysis": downside_analysis,
            "asset_breakdown_test": asset_breakdown,

            "friction_sensitivity_test": friction_sensitivity,
            "cluster_analysis_test": cluster_analysis,
            "multiple_testing_registry": multiple_testing_registry,
            "scientific_conclusion": scientific_conclusion,
            "stop_condition_status": stop_condition_met,

"""Candidate-Specific TimesFM Forecasting Subsystem.

- Primary Architecture: TIMESFM_PRIMARY_5M (12 x 5m steps: +15m, +30m, +45m, +60m).
- Research Architecture: TIMESFM_1M_RESEARCH (60 x 1m steps).
- Full 12-step path persisted in full_path_json.
- Raw quantiles, crossing detection, and post-processing explicitly audited.
- Never labels postprocessed quantiles as 'calibrated' without empirical proof.
- Uncalibrated dispersion labels: DISPERSION_BUCKET_UNCALIBRATED.

_timesfm_singleton = None
_timesfm_singleton_lock = threading.Lock()


def get_timesfm_provider():
    """Lazily load singleton TimesFMProvider if torch and timesfm are present."""
    global _timesfm_singleton
    with _timesfm_singleton_lock:
        if _timesfm_singleton is not None:
            return _timesfm_singleton
        try:
            from .forecast_service import TimesFMProvider
            provider = TimesFMProvider()
            if provider.load_model():
                _timesfm_singleton = provider
                return _timesfm_singleton
        except Exception as e:
            logger.debug("TimesFMProvider could not be loaded: %s", e)
        return None


def generate_candidate_forecast(
    context: CandidateForecastContext,
    historical_candles: List[dict],
    model_override: Optional[Any] = None,
    timeframe_mode: str = TIMESFM_PRIMARY_5M
) -> CandidateForecastResult:
    """Generate and freeze candidate-specific TimesFM future distribution and path.

    Strict Invariants:
    1. Zero Lookahead: Truncates historical context strictly at context.candidate_timestamp_ms.
       Candle is included ONLY if its confirmed completion time <= candidate_timestamp_ms.
    2. Primary 5m Architecture: Default 12 x 5m steps (+15m, +30m, +45m, +60m).
    3. Explicit Quantile Provenance: Persists raw quantiles, detects crossings, tracks sorting.
    4. Uncalibrated Dispersion: Exposes raw dispersion, labels bucket DISPERSION_BUCKET_UNCALIBRATED.
    5. Immutable Persistence: Persists snapshot in candidate_forecasts table.
    """
    t_cutoff = context.candidate_timestamp_ms
    is_5m_mode = (timeframe_mode == TIMESFM_PRIMARY_5M)
    step_duration_ms = 300_000 if is_5m_mode else 60_000
    horizon_steps = TIMESFM_HORIZON_STEPS_5M if is_5m_mode else TIMESFM_HORIZON_STEPS_1M
    source_tf = "5m" if is_5m_mode else "1m"

    # 1. Filter historical bars strictly <= t_cutoff ensuring complete closed candle causality
    causal_bars = []
    for c in historical_candles:
        ts = int(c.get("ts", 0))
        # Determine bar end timestamp: if close_ts is present use it, else calculate from open ts + step duration
        bar_end_ts = int(c.get("close_ts", ts + step_duration_ms))
        if bar_end_ts <= t_cutoff or (ts <= t_cutoff and c.get("closed", True) and bar_end_ts <= t_cutoff + 1000):
            causal_bars.append(c)

    causal_bars.sort(key=lambda x: int(x["ts"]))

    # Snapshot hash computation
    snapshot_hash = hashlib.sha256(
        f"{context.candidate_id}:{t_cutoff}:{round(context.entry_price, 4)}:{context.direction}:{timeframe_mode}".encode()
    ).hexdigest()[:16]

    if len(causal_bars) < 12:
        logger.warning("Insufficient causal context bars (%d < 12) for candidate %s", len(causal_bars), context.candidate_id)
        cur_price = context.entry_price
        path = {"15m": cur_price, "30m": cur_price, "45m": cur_price, "60m": cur_price}
        full_path = [cur_price] * horizon_steps
        raw_q = {"raw_q10": cur_price * 0.99, "raw_q50": cur_price, "raw_q90": cur_price * 1.01}
        post_q = {"q10": cur_price * 0.99, "q50": cur_price, "q90": cur_price * 1.01}
        context_hash = hashlib.sha256(f"INSUFFICIENT_{cur_price}".encode()).hexdigest()[:16]

        return CandidateForecastResult(
            forecast_id=f"FC-{uuid.uuid4().hex[:12]}",
            candidate_id=context.candidate_id,
            symbol=context.symbol,
            canonical_instrument_id=context.canonical_instrument_id,
            data_venue="BINANCE",
            forecast_timestamp=t_cutoff,
            source_timeframe=source_tf,
            timeframe_mode=timeframe_mode,
            path=path,
            full_path=full_path,
            full_path_json=json.dumps(full_path),
            quantiles=post_q,
            raw_quantiles=raw_q,
            quantile_crossing_detected=False,
            postprocessing_method="RAW",
            forecast_return_pct=0.0,
            forecast_dispersion_pct=2.0,
            raw_dispersion_pct=2.0,
            dispersion_bucket="DISPERSION_BUCKET_UNCALIBRATED",
            forecast_range_pct=2.0,
            model="TimesFM_Baseline",
            model_version=TIMESFM_MODEL_VERSION,
            checkpoint=TIMESFM_CHECKPOINT,
            context_length=len(causal_bars),
            direction_prediction="NEUTRAL",
            context_hash=context_hash,
            snapshot_hash=snapshot_hash,
            calibration_status="NOT CALIBRATED",
            data_freshness="INSUFFICIENT_SAMPLE",
            status="SHADOW"
        )

    # 2. Extract causal price history
    closes = np.array([float(b["close"]) for b in causal_bars], dtype=np.float64)
    last_close = closes[-1]
    context_len = min(len(closes), 128)
    sub_closes = closes[-context_len:]
    context_hash = hashlib.sha256(
        json.dumps([round(float(x), 4) for x in sub_closes]).encode()
    ).hexdigest()[:16]

    t_total_start = time.perf_counter()

    # 3. Model Inference (PyTorch TimesFM or Econometric AR Baseline)
    step_predictions: np.ndarray
    raw_model_quantiles = None
    actual_inf_ms = 0.0
    provider_class = "EconometricBaselineProvider"
    model_class = "AutoregressiveMomentumMeanReversion"
    model_name = "Econometric_AR_Baseline"
    checkpoint = "statistical_fallback"
    device = "cpu"
    cache_hit = False

    t_inf_start = time.perf_counter()
    provider = model_override if model_override is not None else get_timesfm_provider()

    if provider is not None and hasattr(provider, "model") and provider.model is not None:
        try:
            output = provider.model.predict(
                context=sub_closes.astype(np.float32),
                horizon=horizon_steps,
                return_quantiles=True,
                sort_quantiles=True
            )
            actual_inf_ms = (time.perf_counter() - t_inf_start) * 1000.0
            provider_class = provider.__class__.__name__
            model_class = provider.model.__class__.__name__
            model_name = "TimesFM_3.0_PyTorch"
            checkpoint = getattr(provider, "checkpoint", TIMESFM_CHECKPOINT)
            device = getattr(provider, "device", "cpu")

            f_raw = output.forecast
            if isinstance(f_raw, np.ndarray):
                step_predictions = f_raw.flatten()[:horizon_steps]
            else:
                step_predictions = np.array(f_raw[:horizon_steps], dtype=np.float64)

            if output.quantiles is not None and isinstance(output.quantiles, np.ndarray):
                q_arr = output.quantiles
                if q_arr.ndim == 3:
                    q_arr = q_arr[0]
                raw_model_quantiles = q_arr
        except Exception as e:
            logger.error("TimesFM model predict error (%s), using econometric AR baseline", e)
            t_fallback_start = time.perf_counter()
            step_predictions = _autoregressive_forecast(sub_closes, horizon_steps)
            actual_inf_ms = (time.perf_counter() - t_fallback_start) * 1000.0
            model_name = "Econometric_AR_Baseline"
            provider_class = "EconometricBaselineProvider"
            model_class = "AutoregressiveMomentumMeanReversion"
            checkpoint = "statistical_fallback"
    elif model_override is not None and hasattr(model_override, "forecast"):
        try:
            step_predictions = model_override.forecast(sub_closes, horizon=horizon_steps)
            actual_inf_ms = (time.perf_counter() - t_inf_start) * 1000.0
            model_name = getattr(model_override, "name", model_override.__class__.__name__)
            provider_class = model_override.__class__.__name__
            model_class = model_override.__class__.__name__
        except Exception as e:
            logger.error("Model override error (%s), using econometric AR baseline", e)
            t_fallback_start = time.perf_counter()
            step_predictions = _autoregressive_forecast(sub_closes, horizon_steps)
            actual_inf_ms = (time.perf_counter() - t_fallback_start) * 1000.0
            model_name = "Econometric_AR_Baseline"
            provider_class = "EconometricBaselineProvider"
            model_class = "AutoregressiveMomentumMeanReversion"
            checkpoint = "statistical_fallback"
    else:
        step_predictions = _autoregressive_forecast(sub_closes, horizon_steps)
        actual_inf_ms = (time.perf_counter() - t_inf_start) * 1000.0
        model_name = "Econometric_AR_Baseline"
        provider_class = "EconometricBaselineProvider"
        model_class = "AutoregressiveMomentumMeanReversion"
        checkpoint = "statistical_fallback"

    full_path_list = [round(float(p), 4) for p in step_predictions]

    # Map step horizons:
    # In 5m mode: Step 3 = +15m, Step 6 = +30m, Step 9 = +45m, Step 12 = +60m
    # In 1m mode: Step 15 = +15m, Step 30 = +30m, Step 45 = +45m, Step 60 = +60m
    if is_5m_mode:
        p15 = float(step_predictions[2])
        p30 = float(step_predictions[5])
        p45 = float(step_predictions[8])
        p60 = float(step_predictions[11])
        step_60_idx = 11
    else:
        p15 = float(step_predictions[14])
        p30 = float(step_predictions[29])
        p45 = float(step_predictions[44])
        p60 = float(step_predictions[59])
        step_60_idx = 59

    # 4. Compute Raw Quantiles & Uncertainty Bounds
    t_post_start = time.perf_counter()
    if raw_model_quantiles is not None and len(raw_model_quantiles) > step_60_idx:
        q_row = raw_model_quantiles[step_60_idx]
        num_q = len(q_row)
        raw_q10 = float(q_row[0])
        raw_q50 = float(q_row[num_q // 2])
        raw_q90 = float(q_row[-1])
    else:
        ret_bars = np.diff(np.log(sub_closes[-20:]))
        sigma_bar = float(np.std(ret_bars)) if len(ret_bars) > 1 else 0.003
        steps_to_60m = 12 if is_5m_mode else 60
        sigma_60m = sigma_bar * math.sqrt(steps_to_60m)

        raw_q50 = float(p60)
        raw_q10 = float(p60 * math.exp(-1.28155 * sigma_60m))
        raw_q90 = float(p60 * math.exp(1.28155 * sigma_60m))

    # Detect Quantile Crossing
    crossing_detected = bool(raw_q10 > raw_q50 or raw_q50 > raw_q90 or raw_q10 > raw_q90)

    # Post-processing (explicitly recorded, never called "calibrated")
    if crossing_detected:
        sorted_quantiles = sorted([raw_q10, raw_q50, raw_q90])
        post_q10 = sorted_quantiles[0]
        post_q50 = sorted_quantiles[1]
        post_q90 = sorted_quantiles[2]
        postprocessing_method = "MONOTONIC_SORTED"
    else:
        post_q10 = raw_q10
        post_q50 = raw_q50
        post_q90 = raw_q90
        postprocessing_method = "RAW"

    postprocessing_ms = (time.perf_counter() - t_post_start) * 1000.0

    forecast_ret_pct = round(((p60 - last_close) / last_close) * 100.0, 4)
    raw_dispersion_pct = round(((post_q90 - post_q10) / max(post_q50, 1e-6)) * 100.0, 4)
    range_pct = round(((max(p15, p30, p45, p60) - min(p15, p30, p45, p60)) / last_close) * 100.0, 4)

    # Direction prediction based on 60m return
    if forecast_ret_pct > 0.15:
        dir_pred = "BULLISH"
    elif forecast_ret_pct < -0.15:
        dir_pred = "BEARISH"
    else:
        dir_pred = "NEUTRAL"

    t_ser_start = time.perf_counter()
    forecast_id = f"FC-{uuid.uuid4().hex[:12]}"
    path_dict = {
        "15m": round(p15, 2),
        "30m": round(p30, 2),
        "45m": round(p45, 2),
        "60m": round(p60, 2),
    }
    raw_quantiles_dict = {
        "raw_q10": round(raw_q10, 2),
        "raw_q50": round(raw_q50, 2),
        "raw_q90": round(raw_q90, 2),
    }
    quantiles_dict = {
        "q10": round(post_q10, 2),
        "q50": round(post_q50, 2),
        "q90": round(post_q90, 2),
    }

    full_path_json = json.dumps(full_path_list)
    serialization_ms = (time.perf_counter() - t_ser_start) * 1000.0
    total_candidate_forecast_ms = (time.perf_counter() - t_total_start) * 1000.0

    result = CandidateForecastResult(
        forecast_id=forecast_id,
        candidate_id=context.candidate_id,
        symbol=context.symbol,
        canonical_instrument_id=context.canonical_instrument_id,
        data_venue="BINANCE",
        forecast_timestamp=t_cutoff,
        source_timeframe=source_tf,
        timeframe_mode=timeframe_mode,
        path=path_dict,
        full_path=full_path_list,
        full_path_json=full_path_json,
        quantiles=quantiles_dict,
        raw_quantiles=raw_quantiles_dict,
        quantile_crossing_detected=crossing_detected,
        postprocessing_method=postprocessing_method,
        forecast_return_pct=forecast_ret_pct,
        forecast_dispersion_pct=raw_dispersion_pct,
        raw_dispersion_pct=raw_dispersion_pct,
        dispersion_bucket="DISPERSION_BUCKET_UNCALIBRATED",
        forecast_range_pct=range_pct,
        model=model_name,
        model_version=TIMESFM_MODEL_VERSION,
        checkpoint=checkpoint,
        context_length=context_len,
        direction_prediction=dir_pred,
        context_hash=context_hash,
        snapshot_hash=snapshot_hash,
        provider_class=provider_class,
        model_class=model_class,
        device=device,
        actual_model_inference_ms=round(actual_inf_ms, 2),
        serialization_ms=round(serialization_ms, 2),
        postprocessing_ms=round(postprocessing_ms, 2),
        compatibility_calculation_ms=0.0,
        total_candidate_forecast_ms=round(total_candidate_forecast_ms, 2),
        cache_hit=cache_hit,
        calibration_status="NOT CALIBRATED",
        data_freshness="FRESH",
        status="SHADOW"
    )

    # 5. Persist immutable record in database
    db_record = {
        "forecast_id": result.forecast_id,
        "candidate_id": result.candidate_id,
        "symbol": result.symbol,
        "canonical_instrument_id": result.canonical_instrument_id,

        "q10": result.quantiles["q10"],

        "raw_q10": result.raw_quantiles["raw_q10"],

        "raw_q90": result.raw_quantiles["raw_q90"],
        "quantile_crossing_detected": 1 if result.quantile_crossing_detected else 0,
        "postprocessed_q10": result.quantiles["q10"],
        "postprocessed_q50": result.quantiles["q50"],
        "postprocessed_q90": result.quantiles["q90"],
        "postprocessing_method": result.postprocessing_method,
        "forecast_return_pct": result.forecast_return_pct,
        "forecast_dispersion_pct": result.forecast_dispersion_pct,
        "dispersion_bucket": result.dispersion_bucket,
        "model_name": result.model,
        "model_version": result.model_version,
        "checkpoint": result.checkpoint,
        "context_length": result.context_length,
        "direction_prediction": result.direction_prediction,
        "context_hash": result.context_hash,
        "snapshot_hash": result.snapshot_hash,
        "provider_class": result.provider_class,

def _autoregressive_forecast(series: np.ndarray, horizon: int) -> np.ndarray:
    """Deterministic, causal autoregressive momentum and mean-reversion forecaster."""
    y = np.copy(series)
    preds = []
    ema = float(np.mean(y[-10:]))
    for _ in range(horizon):
        ret_short = (y[-1] - y[-3]) / y[-3] if len(y) >= 3 else 0.0
        drift = y[-1] * (0.25 * ret_short) + 0.05 * (ema - y[-1])
        next_val = y[-1] + drift
        preds.append(next_val)
        y = np.append(y, next_val)
    return np.array(preds, dtype=np.float64)

# Quantile Calibration & Metric Helpers (Evaluated Out-Of-Sample)
# =====================================================================

def calculate_quantile_coverage(actual_series: List[float], q_series: List[float], quantile: float) -> float:
    """Empirical proportion of observations falling below the predicted quantile: P(y <= q)."""
    if not actual_series or len(actual_series) != len(q_series):
        return 0.0
    n = len(actual_series)
    count_below = sum(1 for y, q in zip(actual_series, q_series) if y <= q)
    return round((count_below / n) * 100.0, 2)


def calculate_interval_coverage(actual_series: List[float], lower_series: List[float], upper_series: List[float]) -> float:
    """Empirical coverage of prediction interval [lower, upper]: P(lower <= y <= upper)."""
    if not actual_series or len(actual_series) != len(lower_series) or len(actual_series) != len(upper_series):
        return 0.0
    n = len(actual_series)
    count_inside = sum(1 for y, l, u in zip(actual_series, lower_series, upper_series) if l <= y <= u)
    return round((count_inside / n) * 100.0, 2)


def calculate_interval_width(lower_series: List[float], upper_series: List[float]) -> float:
    """Mean absolute interval width: E[upper - lower]."""
    if not lower_series or len(lower_series) != len(upper_series):
        return 0.0
    widths = [abs(u - l) for l, u in zip(lower_series, upper_series)]
    return round(float(np.mean(widths)), 4)


def calculate_pinball_loss(actual_series: List[float], q_series: List[float], quantile: float) -> float:
    """Pinball / quantile loss for evaluating quantile forecast accuracy."""
    if not actual_series or len(actual_series) != len(q_series):
        return 0.0
    losses = []
    for y, q in zip(actual_series, q_series):
        diff = y - q
        if diff >= 0:
            losses.append(quantile * diff)
        else:
            losses.append((quantile - 1.0) * diff)
    return round(float(np.mean(losses)), 4)
